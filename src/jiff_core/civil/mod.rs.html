17/// Creates a new `DateTime` value in a `const` context.
18///
19/// This is a convenience free function for [`DateTime::new`] that panics when
20/// the given datetime is not valid. It is intended to provide a terse syntax
21/// for constructing `DateTime` values from parameters that are known to be
22/// valid.
23///
24/// # Panics
25///
26/// This routine panics when [`DateTime::new`] would return an error. That
27/// is, when the given components do not correspond to a valid datetime.
28/// Namely, all of the following must be true:
29///
30/// * The year must be in the range `-9999..=9999`.
31/// * The month must be in the range `1..=12`.
32/// * The day must be at least `1` and must be at most the number of days
33/// in the corresponding month. So for example, `2024-02-29` is valid but
34/// `2023-02-29` is not.
35/// * `0 <= hour <= 23`
36/// * `0 <= minute <= 59`
37/// * `0 <= second <= 59`
38/// * `0 <= subsec_nanosecond <= 999,999,999`
39///
40/// Similarly, when used in a const context, invalid parameters will prevent
41/// your Rust program from compiling.
42///
43/// # Example
44///
45/// ```
46/// use jiff_core::civil::datetime;
47///
48/// let dt = datetime(2024, 2, 29, 21, 30, 5, 123_456_789);
49/// assert_eq!(dt.date().year(), 2024);
50/// assert_eq!(dt.date().month(), 2);
51/// assert_eq!(dt.date().day(), 29);
52/// assert_eq!(dt.time().hour(), 21);
53/// assert_eq!(dt.time().minute(), 30);
54/// assert_eq!(dt.time().second(), 5);
55/// assert_eq!(dt.time().millisecond(), 123);
56/// assert_eq!(dt.time().microsecond(), 456);
57/// assert_eq!(dt.time().nanosecond(), 789);
58/// ```
59#[inline]
60pub const fn datetime(
61    year: i16,
62    month: i8,
63    day: i8,
64    hour: i8,
65    minute: i8,
66    second: i8,
67    subsec_nanosecond: i32,
68) -> DateTime {
69    unwrapr!(
70        DateTime::new(
71            year,
72            month,
73            day,
74            hour,
75            minute,
76            second,
77            subsec_nanosecond,
78        ),
79        "invalid datetime"
80    )
81}

167/// Returns true if and only if the given year is a leap year.
168///
169/// A leap year is a year with 366 days. Non-leap years have 365 days.
170#[inline]
171pub const fn is_leap_year(year: i16) -> bool {
172    // From: https://github.com/BurntSushi/jiff/pull/23
173    let d = if year % 25 != 0 { 4 } else { 16 };
174    (year % d) == 0
175}

177/// Return the number of days in the given year.
178///
179/// This is guaranteed to either return `365` or `366`.
180#[inline]
181pub const fn days_in_year(year: i16) -> i16 {
182    if is_leap_year(year) {
183        366
184    } else {
185        365
186    }
187}

189/// Return the number of days in the given month.
190///
191/// This is guaranteed to return a value in the range `1..=31`.
192#[inline]
193pub const fn days_in_month(year: i16, month: i8) -> i8 {
194    // From: https://github.com/BurntSushi/jiff/pull/23
195    if month == 2 {
196        if is_leap_year(year) {
197            29
198        } else {
199            28
200        }
201    } else {
202        30 | (month ^ month >> 3)
203    }
204}

206/// Returns true if the given ISO 8601 week date year is a "long" year or not.
207///
208/// A "long" year is a year with 53 weeks. Otherwise, it's a "short" year
209/// with 52 weeks.
210#[inline]
211pub const fn is_long_iso_week_year(year: i16) -> bool {
212    // Inspired by: https://en.wikipedia.org/wiki/ISO_week_date#Weeks_per_year
213    let last =
214        unwrapr!(Date::new(year, 12, 31), "last day of year is always valid");
215    let weekday = last.weekday();
216    matches!(weekday, Weekday::Thursday)
217        || (last.in_leap_year() && matches!(weekday, Weekday::Friday))
218}
219
220/// Returns the total number of weeks in the year of this ISO 8601 week
221/// date.
222///
223/// It is guaranteed that the value returned is either 52 or 53. The
224/// latter case occurs precisely when [`ISOWeekDate::in_long_year`]
225/// returns `true`.
226#[inline]
227pub const fn weeks_in_iso_week_year(year: i16) -> i8 {
228    if is_long_iso_week_year(year) {
229        53
230    } else {
231        52
232    }
233}
