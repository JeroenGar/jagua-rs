6/// Sort a slice lexicographically by two integer‐like keys, then by a comparator.
7///
8/// This is the most general API: primary key, secondary key,
9/// and a tiebreaking comparator.
10///
11/// - Accepts `parallel: bool`. Ignored without `allow_multithreading`.
12/// - Two variants: allocate internally or reuse a buffer.
13///
14/// # Examples
15/// ```
16/// use i_key_sort::sort::two_keys_cmp::TwoKeysAndCmpSort;
17///
18/// let mut v = vec![(1u32,0i32,9i32), (1,0,3), (1,1,1)];
19/// v.sort_by_two_keys_then_by(true, |x| x.0, |x| x.1, |a,b| a.2.cmp(&b.2));
20/// assert_eq!(v, [(1,0,3), (1,0,9), (1,1,1)]);
21/// ```
22#[cfg(not(feature = "allow_multithreading"))]
23pub trait TwoKeysAndCmpSort<T> {
24    fn sort_by_two_keys_then_by<K1, K2, F1, F2, F3>(
25        &mut self,
26        parallel: bool,
27        key1: F1,
28        key2: F2,
29        compare: F3,
30    ) where
31        K1: SortKey,
32        K2: SortKey,
33        F1: KeyFn<T, K1>,
34        F2: KeyFn<T, K2>,
35        F3: CmpFn<T>;
36
37    fn sort_by_two_keys_then_by_and_buffer<K1, K2, F1, F2, F3>(
38        &mut self,
39        parallel: bool,
40        reusable_buffer: &mut Vec<T>,
41        key1: F1,
42        key2: F2,
43        compare: F3,
44    ) where
45        K1: SortKey,
46        K2: SortKey,
47        F1: KeyFn<T, K1>,
48        F2: KeyFn<T, K2>,
49        F3: CmpFn<T>;
50}

200#[inline]
201pub(crate) fn sort_unstable_by_two_keys_then_by<T, K1, K2, F1, F2, F3>(
202    slice: &mut [T],
203    key1: F1,
204    key2: F2,
205    compare: F3,
206) where
207    K1: SortKey,
208    K2: SortKey,
209    F1: KeyFn<T, K1>,
210    F2: KeyFn<T, K2>,
211    F3: CmpFn<T>,
212{
213    slice.sort_unstable_by(|a, b| {
214        key1(a)
215            .cmp(&key1(b))
216            .then(key2(a).cmp(&key2(b)))
217            .then(compare(a, b))
218    });
219}

263    fn test(count: usize) {
264        let mut org: Vec<_> = reversed_2d_array(count);
265        let mut arr1 = org.clone();
266        let mut arr2 = org.clone();
267        arr1.sort_by_two_keys_then_by(true, |a| a.0, |a| a.1, |a, b| a.2.cmp(&b.2));
268        arr2.sort_by_two_keys_then_by(false, |a| a.0, |a| a.1, |a, b| a.2.cmp(&b.2));
269
270        org.sort_unstable_by(|a, b| a.0.cmp(&b.0).then(a.1.cmp(&b.1)).then(a.2.cmp(&b.2)));
271        assert!(arr1 == org);
272        assert!(arr2 == org);
273    }

275    fn reversed_2d_array(count: usize) -> Vec<(u32, i32, i32)> {
276        let mut arr = Vec::with_capacity(count * count * count);
277        for i in (0..count as u32).rev() {
278            for x in (0..count as i32).rev() {
279                for y in (0..count as i32).rev() {
280                    arr.push((i, x, y))
281                }
282            }
283        }
284
285        arr
286    }
