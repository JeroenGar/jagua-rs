1use crate::sort::bin_layout::BIN_SORT_MIN;
2use crate::sort::buffer::{CopyFromNotOverlap, CopyNotOverlapValue, DoubleRangeSlices};
3use crate::sort::key::{KeyFn, SortKey};
4use crate::sort::mapper::Mapper;

7impl Mapper {
8    #[inline]
9    pub(crate) fn sort_chunks_by_one_key<K: SortKey, T: Copy, F: KeyFn<T, K>>(
10        &self,
11        src: &mut [T],
12        buf: &mut [T],
13        key: F,
14        copy_to_src: bool,
15    ) {
16        const TINY_SORT_MAX: usize = BIN_SORT_MIN;
17
18        // if `copy_to_src` is true
19        // must copy `src` to `buf`, since the result array is in the buffer
20
21        for chunk in self.iter() {
22            let range = chunk.as_range();
23            match range.len() {
24                0 => continue,
25                1 => {
26                    if copy_to_src {
27                        src.copy_value_from(buf, range.start);
28                    }
29                }
30                2..TINY_SORT_MAX => {
31                    // SAFETY: mapper ranges never overlap; (src, buf) are distinct buffers.
32                    let sub_buf = unsafe { buf.get_unchecked_mut(range.clone()) };
33                    sub_buf.sort_unstable_by_key(key);
34
35                    if copy_to_src {
36                        src.copy_to_range_from_not_overlap(sub_buf, range);
37                    }
38                }
39                _ => {
40                    let (sub_src, sub_buf) = range.mut_slices(src, buf);
41                    sub_buf.ser_sort_by_one_key_and_buffer(sub_src, key, !copy_to_src);
42                }
43            }
44        }
45    }
