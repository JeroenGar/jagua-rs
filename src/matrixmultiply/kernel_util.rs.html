1/// If the microkernel tile is a square, mr == nr,
2/// then we can change the problem from C = A B to C^T = B^T A^T
3///
4/// Whenever rsc == 1, we switch the orientation around.
5/// *Any contiguous positive stride* C will exit this function with rsc arbitrary, csc == 1.
6///
7/// Input row stride (rs), column stride (cs) and outcomes
8/// rsc     csc     explanation
9/// M       1       no change => (M, 1)
10/// 1       Z       transpose => (Z, 1)
11/// M1      M2      no change => (M1, M2)
12///
13/// Where M signifies number M != 1 and Z any number (pos, neg, or zero)
14///
15/// + `mr`, `nr` kernel size; must be square
16/// + `a` pointer to packing buffer
17/// + `b` pointer to packing buffer
18/// + `rsc` row stride of c
19/// + `csc` colum stride of c
20#[inline(always)]
21#[allow(unused)]
22pub(crate) fn preferential_transpose<T>(mr: usize, nr: usize, a: *const T, b: *const T, rsc: isize, csc: isize)
23    -> (*const T, *const T, isize, isize)
24{
25    debug_assert_eq!(mr, nr, "Transpose requires that MR == NR");
26    // Compute C in whichever orientation makes the output columns contiguous
27    let prefer_row_major_c = rsc != 1;
28    if prefer_row_major_c { (a, b, rsc, csc) } else { (b, a, csc, rsc) }
29}

31/// Dereference pointer at offset `i`
32#[inline(always)]
33pub(crate) unsafe fn at<T: Copy>(ptr: *const T, i: usize) -> T {
34    *ptr.add(i)
35}
