44/// Detect which implementation to use and select it using the selector's
45/// .select(Kernel) method.
46///
47/// This function is called one or more times during a whole program's
48/// execution, it may be called for each gemm kernel invocation or fewer times.
49#[inline]
50pub(crate) fn detect<G>(selector: G) where G: GemmSelect<T> {
51    // dispatch to specific compiled versions
52    #[cfg(any(target_arch="x86", target_arch="x86_64"))]
53    {
54        #[cfg(has_avx512)]
55        {
56            if is_x86_feature_detected_!("avx512f") {
57                return selector.select(KernelAvx512);
58            }
59        }
60        if is_x86_feature_detected_!("fma") && is_x86_feature_detected_!("avx2") {
61            return selector.select(KernelFmaAvx2);
62        } else if is_x86_feature_detected_!("avx") {
63            return selector.select(KernelAvx);
64        }
65    }
66    #[cfg(target_arch="aarch64")]
67    {
68        if is_aarch64_feature_detected_!("neon") {
69            return selector.select(KernelNeon);
70        }
71    }
72    #[cfg(all(target_arch="wasm32", target_feature="simd128"))]
73    {
74        return selector.select(KernelWasmSimd);
75    }
76    #[allow(unreachable_code)]
77    return selector.select(KernelFallback);
78}

307#[inline(always)]
308#[cfg(any(target_arch="x86", target_arch="x86_64"))]
309unsafe fn kernel_x86_avx<MA>(k: usize, alpha: T, a: *const T, b: *const T,
310                             beta: T, c: *mut T, rsc: isize, csc: isize)
311    where MA: SMultiplyAdd,
312{
313    const MR: usize = KernelAvx::MR;
314    const NR: usize = KernelAvx::NR;
315
316    debug_assert_ne!(k, 0);
317
318    let mut ab = [_mm256_setzero_ps(); MR];
319
320    // Compute C in whichever orientation makes the output columns contiguous
321    let (mut a, mut b, rsc, csc) = preferential_transpose(MR, NR, a, b, rsc, csc);
322
323    macro_rules! permute_mask {
324        ($z:expr, $y:expr, $x:expr, $w:expr) => {
325            ($z << 6) | ($y << 4) | ($x << 2) | $w
326        }
327    }
328
329    // Start data load before each iteration
330    let mut av = _mm256_load_ps(a);
331    let mut bvl = _mm256_broadcast_ps(&*(b.add(0) as *const _));
332    let mut bvh = _mm256_broadcast_ps(&*(b.add(4) as *const _));
333
334    // Compute A B
335    unroll_by_with_last!(4 => k, is_last, {
336        // We compute abij = ai bj
337        //
338        //
339        //   ab0:    ab1:    ab2:    ab3:
340        // ( ab00  ( ab10  ( ab20  ( ab30
341        //   ab11    ab21    ab31    ab01
342        //   ab22    ab32    ab02    ab12
343        //   ab33    ab03    ab13    ab23
344        //   ab40    ab50    ab60    ab70
345        //   ab51    ab61    ab71    ab41
346        //   ab62    ab72    ab42    ab52
347        //   ab73 )  ab43 )  ab53 )  ab63 )
348        //
349        // ab1357: ab3175: ab5713: ab7531:
350        // ( ab04  ( ab14  ( ab24  ( ab34
351        //   ab15    ab25    ab35    ab05
352        //   ab26    ab36    ab06    ab16
353        //   ab37    ab07    ab17    ab27
354        //   ab44    ab54    ab64    ab74
355        //   ab55    ab65    ab75    ab45
356        //   ab66    ab76    ab46    ab56
357        //   ab77 )  ab47 )  ab57 )  ab67 )
358
359        let a01234567 = av;
360        let a12305674 = _mm256_permute_ps(av, permute_mask!(0, 3, 2, 1));
361        let a23016745 = _mm256_permute_ps(av, permute_mask!(1, 0, 3, 2));
362        let a30127456 = _mm256_permute_ps(av, permute_mask!(2, 1, 0, 3));
363
364        ab[0] = MA::multiply_add(a01234567, bvl, ab[0]);
365        ab[4] = MA::multiply_add(a01234567, bvh, ab[4]);
366
367        ab[1] = MA::multiply_add(a12305674, bvl, ab[1]);
368        ab[5] = MA::multiply_add(a12305674, bvh, ab[5]);
369
370        ab[2] = MA::multiply_add(a23016745, bvl, ab[2]);
371        ab[6] = MA::multiply_add(a23016745, bvh, ab[6]);
372
373        ab[3] = MA::multiply_add(a30127456, bvl, ab[3]);
374        ab[7] = MA::multiply_add(a30127456, bvh, ab[7]);
375
376        if !is_last {
377            a = a.add(MR);
378            b = b.add(NR);
379
380            bvl = _mm256_broadcast_ps(&*(b.add(0) as *const _));
381            bvh = _mm256_broadcast_ps(&*(b.add(4) as *const _));
382            av = _mm256_load_ps(a);
383        }
384    });
385
386    let alphav = _mm256_set1_ps(alpha);
387
388    // Permute to put the abij elements in order
389    let t0 = ab[0];
390    let t1 = ab[1];
391    let t2 = ab[2];
392    let t3 = ab[3];
393
394    let (t0, t1, t2, t3) = (
395        _mm256_blend_ps(t0, t3, 0b10101010),
396        _mm256_blend_ps(t1, t0, 0b10101010),
397        _mm256_blend_ps(t2, t1, 0b10101010),
398        _mm256_blend_ps(t3, t2, 0b10101010),
399    );
400
401    let (t0, t1, t2, t3) = (
402        _mm256_blend_ps(t0, t2, 0b11001100),
403        _mm256_blend_ps(t1, t3, 0b11001100),
404        _mm256_blend_ps(t2, t0, 0b11001100),
405        _mm256_blend_ps(t3, t1, 0b11001100),
406    );
407
408    let t4 = ab[4];
409    let t5 = ab[5];
410    let t6 = ab[6];
411    let t7 = ab[7];
412
413    let (t4, t5, t6, t7) = (
414        _mm256_blend_ps(t4, t7, 0b10101010),
415        _mm256_blend_ps(t5, t4, 0b10101010),
416        _mm256_blend_ps(t6, t5, 0b10101010),
417        _mm256_blend_ps(t7, t6, 0b10101010),
418    );
419
420    let (t4, t5, t6, t7) = (
421        _mm256_blend_ps(t4, t6, 0b11001100),
422        _mm256_blend_ps(t5, t7, 0b11001100),
423        _mm256_blend_ps(t6, t4, 0b11001100),
424        _mm256_blend_ps(t7, t5, 0b11001100),
425    );
426
427    ab[0] = _mm256_permute2f128_ps(t0, t4, 0x20);
428    ab[1] = _mm256_permute2f128_ps(t1, t5, 0x20);
429    ab[2] = _mm256_permute2f128_ps(t2, t6, 0x20);
430    ab[3] = _mm256_permute2f128_ps(t3, t7, 0x20);
431    ab[4] = _mm256_permute2f128_ps(t0, t4, 0x31);
432    ab[5] = _mm256_permute2f128_ps(t1, t5, 0x31);
433    ab[6] = _mm256_permute2f128_ps(t2, t6, 0x31);
434    ab[7] = _mm256_permute2f128_ps(t3, t7, 0x31);
435
436    // Compute α (A B)
437    // Compute here if we don't have fma, else pick up α further down
438    if !MA::IS_FUSED {
439        loop_m!(i, ab[i] = _mm256_mul_ps(alphav, ab[i]));
440    }
441
442    macro_rules! c {
443        ($i:expr, $j:expr) => (c.offset(rsc * $i as isize + csc * $j as isize));
444    }
445
446    // C ← α A B + β C
447    let mut cv = [_mm256_setzero_ps(); MR];
448    if beta != 0. {
449        let betav = _mm256_set1_ps(beta);
450        // Read C
451        if csc == 1 {
452            loop_m!(i, cv[i] = _mm256_loadu_ps(c![i, 0]));
453        } else {
454            loop_m!(i, cv[i] = _mm256_setr_ps(*c![i, 0], *c![i, 1], *c![i, 2], *c![i, 3],
455                                              *c![i, 4], *c![i, 5], *c![i, 6], *c![i, 7]));
456        }
457        // Compute β C
458        loop_m!(i, cv[i] = _mm256_mul_ps(cv[i], betav));
459    }
460
461    // Compute (α A B) + (β C)
462    if !MA::IS_FUSED {
463        loop_m!(i, cv[i] = _mm256_add_ps(cv[i], ab[i]));
464    } else {
465        loop_m!(i, cv[i] = MA::multiply_add(alphav, ab[i], cv[i]));
466    }
467
468    // Store C back to memory
469    if csc == 1 {
470        loop_m!(i, _mm256_storeu_ps(c![i, 0], cv[i]));
471    } else {
472        // Permute to bring each element in the vector to the front and store
473        loop_m!(i, {
474            let cvlo = _mm256_extractf128_ps(cv[i], 0);
475            let cvhi = _mm256_extractf128_ps(cv[i], 1);
476
477            _mm_store_ss(c![i, 0], cvlo);
478            let cperm = _mm_permute_ps(cvlo, permute_mask!(0, 3, 2, 1));
479            _mm_store_ss(c![i, 1], cperm);
480            let cperm = _mm_permute_ps(cperm, permute_mask!(0, 3, 2, 1));
481            _mm_store_ss(c![i, 2], cperm);
482            let cperm = _mm_permute_ps(cperm, permute_mask!(0, 3, 2, 1));
483            _mm_store_ss(c![i, 3], cperm);
484
485            _mm_store_ss(c![i, 4], cvhi);
486            let cperm = _mm_permute_ps(cvhi, permute_mask!(0, 3, 2, 1));
487            _mm_store_ss(c![i, 5], cperm);
488            let cperm = _mm_permute_ps(cperm, permute_mask!(0, 3, 2, 1));
489            _mm_store_ss(c![i, 6], cperm);
490            let cperm = _mm_permute_ps(cperm, permute_mask!(0, 3, 2, 1));
491            _mm_store_ss(c![i, 7], cperm);
492        });
493    }
494}

496// no inline for unmasked kernels
497#[cfg(has_avx512)]
498#[target_feature(enable="avx512f")]
499unsafe fn kernel_target_avx512(k: usize, alpha: T, a: *const T, b: *const T,
500                               beta: T, c: *mut T, rsc: isize, csc: isize)
501{
502    const MR: usize = KernelAvx512::MR;
503    const NR: usize = KernelAvx512::NR;
504    debug_assert_ne!(k, 0);
505
506    let mut ab = [_mm512_setzero_ps(); MR];
507
508    // Compute C in whichever orientation makes the output columns contiguous
509    let (mut a, mut b, rsc, csc) = preferential_transpose(MR, NR, a, b, rsc, csc);
510
511    // Compute A B. The packed buffers are 64-byte aligned
512    let mut bv = _mm512_load_ps(b);
513    unroll_by_with_last!(4 => k, is_last, {
514        loop16!(i, ab[i] = _mm512_fmadd_ps(_mm512_set1_ps(*a.add(i)), bv, ab[i]));
515        if !is_last {
516            a = a.add(MR);
517            b = b.add(NR);
518            bv = _mm512_load_ps(b);
519        }
520    });
521
522    macro_rules! c {
523        ($i:expr, $j:expr) => (c.offset(rsc * $i as isize + csc * $j as isize));
524    }
525
526    // C <- alpha (A B) + beta C, in a single epilogue pass
527    // Fold alpha into the final FMA
528    // When beta == 0 the kernel must not read C
529    let alphav = _mm512_set1_ps(alpha);
530    if beta != 0. {
531        let betav = _mm512_set1_ps(beta);
532        if csc == 1 {
533            loop16!(i, {
534                let cv = _mm512_mul_ps(_mm512_loadu_ps(c![i, 0]), betav);
535                _mm512_storeu_ps(c![i, 0], _mm512_fmadd_ps(alphav, ab[i], cv));
536            });
537        } else {
538            loop16!(i, {
539                let mut tmp = [0.; NR];
540                for j in 0..NR { tmp[j] = *c![i, j]; }
541                let cv = _mm512_mul_ps(_mm512_loadu_ps(tmp.as_ptr()), betav);
542                _mm512_storeu_ps(tmp.as_mut_ptr(), _mm512_fmadd_ps(alphav, ab[i], cv));
543                for j in 0..NR { *c![i, j] = tmp[j]; }
544            });
545        }
546    } else {
547        if csc == 1 {
548            loop16!(i, _mm512_storeu_ps(c![i, 0], _mm512_mul_ps(alphav, ab[i])));
549        } else {
550            loop16!(i, {
551                let mut tmp = [0.; NR];
552                _mm512_storeu_ps(tmp.as_mut_ptr(), _mm512_mul_ps(alphav, ab[i]));
553                for j in 0..NR { *c![i, j] = tmp[j]; }
554            });
555        }
556    }
557}

561unsafe fn kernel_target_neon(k: usize, alpha: T, a: *const T, b: *const T,
562                             beta: T, c: *mut T, rsc: isize, csc: isize)
563{
564    use core::arch::aarch64::*;
565    const MR: usize = KernelNeon::MR;
566    const NR: usize = KernelNeon::NR;
567
568    let (mut a, mut b, rsc, csc) = preferential_transpose(MR, NR, a, b, rsc, csc);
569
570    // Kernel 8 x 8 (a x b)
571    // Four quadrants of 4 x 4
572    let mut ab11 = [vmovq_n_f32(0.); 4];
573    let mut ab12 = [vmovq_n_f32(0.); 4];
574    let mut ab21 = [vmovq_n_f32(0.); 4];
575    let mut ab22 = [vmovq_n_f32(0.); 4];
576
577    // Compute
578    // ab_ij = a_i * b_j for all i, j
579    macro_rules! ab_ij_equals_ai_bj {
580        ($dest:ident, $av:expr, $bv:expr) => {
581            $dest[0] = vfmaq_laneq_f32($dest[0], $bv, $av, 0);
582            $dest[1] = vfmaq_laneq_f32($dest[1], $bv, $av, 1);
583            $dest[2] = vfmaq_laneq_f32($dest[2], $bv, $av, 2);
584            $dest[3] = vfmaq_laneq_f32($dest[3], $bv, $av, 3);
585        }
586    }
587
588    for _ in 0..k {
589        let a1 = vld1q_f32(a);
590        let b1 = vld1q_f32(b);
591        let a2 = vld1q_f32(a.add(4));
592        let b2 = vld1q_f32(b.add(4));
593
594        // compute an outer product ab = a (*) b in four quadrants ab11, ab12, ab21, ab22
595
596        // ab11: [a1 a2 a3 a4] (*) [b1 b2 b3 b4]
597        // ab11: a1b1 a1b2 a1b3 a1b4
598        //       a2b1 a2b2 a2b3 a2b4
599        //       a3b1 a3b2 a3b3 a3b4
600        //       a4b1 a4b2 a4b3 a4b4
601        //  etc
602        ab_ij_equals_ai_bj!(ab11, a1, b1);
603        ab_ij_equals_ai_bj!(ab12, a1, b2);
604        ab_ij_equals_ai_bj!(ab21, a2, b1);
605        ab_ij_equals_ai_bj!(ab22, a2, b2);
606
607        a = a.add(MR);
608        b = b.add(NR);
609    }
610
611    macro_rules! c {
612        ($i:expr, $j:expr) => (c.offset(rsc * $i as isize + csc * $j as isize));
613    }
614
615    // ab *= alpha
616    loop4!(i, ab11[i] = vmulq_n_f32(ab11[i], alpha));
617    loop4!(i, ab12[i] = vmulq_n_f32(ab12[i], alpha));
618    loop4!(i, ab21[i] = vmulq_n_f32(ab21[i], alpha));
619    loop4!(i, ab22[i] = vmulq_n_f32(ab22[i], alpha));
620
621    // load one float32x4_t from four pointers
622    macro_rules! loadq_from_pointers {
623        ($p0:expr, $p1:expr, $p2:expr, $p3:expr) => (
624            {
625                let v = vld1q_dup_f32($p0);
626                let v = vld1q_lane_f32($p1, v, 1);
627                let v = vld1q_lane_f32($p2, v, 2);
628                let v = vld1q_lane_f32($p3, v, 3);
629                v
630            }
631        );
632    }
633
634    if beta != 0. {
635        // load existing value in C
636        let mut c11 = [vmovq_n_f32(0.); 4];
637        let mut c12 = [vmovq_n_f32(0.); 4];
638        let mut c21 = [vmovq_n_f32(0.); 4];
639        let mut c22 = [vmovq_n_f32(0.); 4];
640
641        if csc == 1 {
642            loop4!(i, c11[i] = vld1q_f32(c![i + 0, 0]));
643            loop4!(i, c12[i] = vld1q_f32(c![i + 0, 4]));
644            loop4!(i, c21[i] = vld1q_f32(c![i + 4, 0]));
645            loop4!(i, c22[i] = vld1q_f32(c![i + 4, 4]));
646        } else {
647            loop4!(i, c11[i] = loadq_from_pointers!(c![i + 0, 0], c![i + 0, 1], c![i + 0, 2], c![i + 0, 3]));
648            loop4!(i, c12[i] = loadq_from_pointers!(c![i + 0, 4], c![i + 0, 5], c![i + 0, 6], c![i + 0, 7]));
649            loop4!(i, c21[i] = loadq_from_pointers!(c![i + 4, 0], c![i + 4, 1], c![i + 4, 2], c![i + 4, 3]));
650            loop4!(i, c22[i] = loadq_from_pointers!(c![i + 4, 4], c![i + 4, 5], c![i + 4, 6], c![i + 4, 7]));
651        }
652
653        let betav = vmovq_n_f32(beta);
654
655        // ab += β C
656        loop4!(i, ab11[i] = vfmaq_f32(ab11[i], c11[i], betav));
657        loop4!(i, ab12[i] = vfmaq_f32(ab12[i], c12[i], betav));
658        loop4!(i, ab21[i] = vfmaq_f32(ab21[i], c21[i], betav));
659        loop4!(i, ab22[i] = vfmaq_f32(ab22[i], c22[i], betav));
660    }
661
662    // c <- ab
663    // which is in full
664    //   C <- α A B (+ β C)
665    if csc == 1 {
666        loop4!(i, vst1q_f32(c![i + 0, 0], ab11[i]));
667        loop4!(i, vst1q_f32(c![i + 0, 4], ab12[i]));
668        loop4!(i, vst1q_f32(c![i + 4, 0], ab21[i]));
669        loop4!(i, vst1q_f32(c![i + 4, 4], ab22[i]));
670    } else {
671        loop4!(i, vst1q_lane_f32(c![i + 0, 0], ab11[i], 0));
672        loop4!(i, vst1q_lane_f32(c![i + 0, 1], ab11[i], 1));
673        loop4!(i, vst1q_lane_f32(c![i + 0, 2], ab11[i], 2));
674        loop4!(i, vst1q_lane_f32(c![i + 0, 3], ab11[i], 3));
675
676        loop4!(i, vst1q_lane_f32(c![i + 0, 4], ab12[i], 0));
677        loop4!(i, vst1q_lane_f32(c![i + 0, 5], ab12[i], 1));
678        loop4!(i, vst1q_lane_f32(c![i + 0, 6], ab12[i], 2));
679        loop4!(i, vst1q_lane_f32(c![i + 0, 7], ab12[i], 3));
680
681        loop4!(i, vst1q_lane_f32(c![i + 4, 0], ab21[i], 0));
682        loop4!(i, vst1q_lane_f32(c![i + 4, 1], ab21[i], 1));
683        loop4!(i, vst1q_lane_f32(c![i + 4, 2], ab21[i], 2));
684        loop4!(i, vst1q_lane_f32(c![i + 4, 3], ab21[i], 3));
685
686        loop4!(i, vst1q_lane_f32(c![i + 4, 4], ab22[i], 0));
687        loop4!(i, vst1q_lane_f32(c![i + 4, 5], ab22[i], 1));
688        loop4!(i, vst1q_lane_f32(c![i + 4, 6], ab22[i], 2));
689        loop4!(i, vst1q_lane_f32(c![i + 4, 7], ab22[i], 3));
690    }
691}

693#[cfg(all(target_arch="wasm32", target_feature="simd128"))]
694unsafe fn kernel_target_wasm_simd(k: usize, alpha: T, a: *const T, b: *const T,
695                                  beta: T, c: *mut T, rsc: isize, csc: isize)
696{
697    use core::arch::wasm32::*;
698    const MR: usize = KernelWasmSimd::MR;
699    const NR: usize = KernelWasmSimd::NR;
700
701    // Use f32x4_relaxed_madd when enabled
702    // by spec relaxed_madd is a fused multiply-add when possible, otherwise it is a multiply then add
703    #[cfg(target_feature = "relaxed-simd")]
704    #[inline(always)]
705    unsafe fn muladd(a: v128, b: v128, c: v128) -> v128 {
706        f32x4_relaxed_madd(a, b, c)
707    }
708
709    #[cfg(not(target_feature = "relaxed-simd"))]
710    #[inline(always)]
711    unsafe fn muladd(a: v128, b: v128, c: v128) -> v128 {
712        f32x4_add(f32x4_mul(a, b), c)
713    }
714
715    let (mut a, mut b, rsc, csc) = preferential_transpose(MR, NR, a, b, rsc, csc);
716
717    // Kernel 8 x 8 (a x b)
718    // Four quadrants of 4 x 4
719    let zero = f32x4_splat(0.);
720    let mut ab11 = [zero; 4];
721    let mut ab12 = [zero; 4];
722    let mut ab21 = [zero; 4];
723    let mut ab22 = [zero; 4];
724
725    /// multiple sets of ptr => v1, v2 supported
726    macro_rules! load_vectors {
727        ($($a:ident => $a1:ident, $a2:ident),+) => {
728            $(
729                $a1 = v128_load($a as *const v128);
730                $a2 = v128_load($a.add(4) as *const v128);
731            )+
732        }
733    }
734
735    macro_rules! splat_lane {
736        ($v:expr, $n:tt) =>  { i32x4_shuffle::<$n, $n, $n, $n>($v, $v) };
737    }
738
739    // ab_ij = a_i * b_j for all i, j
740    macro_rules! ab_ij_equals_ai_bj {
741        ($dest:ident, $av:expr, $bv:expr) => {
742            $dest[0] = muladd($bv, splat_lane!($av, 0), $dest[0]);
743            $dest[1] = muladd($bv, splat_lane!($av, 1), $dest[1]);
744            $dest[2] = muladd($bv, splat_lane!($av, 2), $dest[2]);
745            $dest[3] = muladd($bv, splat_lane!($av, 3), $dest[3]);
746        }
747    }
748
749    let (mut a1, mut b1, mut a2, mut b2);
750
751    for _ in 0..k {
752        load_vectors!(a => a1, a2, b => b1, b2);
753        ab_ij_equals_ai_bj!(ab11, a1, b1);
754        ab_ij_equals_ai_bj!(ab12, a1, b2);
755        ab_ij_equals_ai_bj!(ab21, a2, b1);
756        ab_ij_equals_ai_bj!(ab22, a2, b2);
757        a = a.add(MR);
758        b = b.add(NR);
759    }
760
761    macro_rules! c {
762        ($i:expr, $j:expr) => (c.offset(rsc * $i as isize + csc * $j as isize));
763    }
764
765    // ab *= alpha
766    let alphav = f32x4_splat(alpha);
767    loop4!(i, ab11[i] = f32x4_mul(ab11[i], alphav));
768    loop4!(i, ab12[i] = f32x4_mul(ab12[i], alphav));
769    loop4!(i, ab21[i] = f32x4_mul(ab21[i], alphav));
770    loop4!(i, ab22[i] = f32x4_mul(ab22[i], alphav));
771
772    // load one v128 from four pointers
773    macro_rules! loadq_from_pointers {
774        ($p0:expr, $p1:expr, $p2:expr, $p3:expr) => ({
775            let v = f32x4_splat(0.);
776            let v = v128_load32_lane::<0>(v, $p0 as *const u32);
777            let v = v128_load32_lane::<1>(v, $p1 as *const u32);
778            let v = v128_load32_lane::<2>(v, $p2 as *const u32);
779            let v = v128_load32_lane::<3>(v, $p3 as *const u32);
780            v
781        });
782    }
783
784    if beta != 0. {
785        // load existing value in C
786        let mut c11 = [zero; 4];
787        let mut c12 = [zero; 4];
788        let mut c21 = [zero; 4];
789        let mut c22 = [zero; 4];
790
791        if csc == 1 {
792            loop4!(i, c11[i] = v128_load(c![i + 0, 0] as *const v128));
793            loop4!(i, c12[i] = v128_load(c![i + 0, 4] as *const v128));
794            loop4!(i, c21[i] = v128_load(c![i + 4, 0] as *const v128));
795            loop4!(i, c22[i] = v128_load(c![i + 4, 4] as *const v128));
796        } else {
797            loop4!(i, c11[i] = loadq_from_pointers!(c![i + 0, 0], c![i + 0, 1], c![i + 0, 2], c![i + 0, 3]));
798            loop4!(i, c12[i] = loadq_from_pointers!(c![i + 0, 4], c![i + 0, 5], c![i + 0, 6], c![i + 0, 7]));
799            loop4!(i, c21[i] = loadq_from_pointers!(c![i + 4, 0], c![i + 4, 1], c![i + 4, 2], c![i + 4, 3]));
800            loop4!(i, c22[i] = loadq_from_pointers!(c![i + 4, 4], c![i + 4, 5], c![i + 4, 6], c![i + 4, 7]));
801        }
802
803        let betav = f32x4_splat(beta);
804        // ab += β C
805        loop4!(i, ab11[i] = muladd(betav, c11[i], ab11[i]));
806        loop4!(i, ab12[i] = muladd(betav, c12[i], ab12[i]));
807        loop4!(i, ab21[i] = muladd(betav, c21[i], ab21[i]));
808        loop4!(i, ab22[i] = muladd(betav, c22[i], ab22[i]));
809    }
810
811    // c <- ab
812    // which is in full
813    //   C <- α A B (+ β C)
814    if csc == 1 {
815        loop4!(i, v128_store(c![i + 0, 0] as *mut v128, ab11[i]));
816        loop4!(i, v128_store(c![i + 0, 4] as *mut v128, ab12[i]));
817        loop4!(i, v128_store(c![i + 4, 0] as *mut v128, ab21[i]));
818        loop4!(i, v128_store(c![i + 4, 4] as *mut v128, ab22[i]));
819    } else {
820        loop4!(i, v128_store32_lane::<0>(ab11[i], c![i + 0, 0] as *mut u32));
821        loop4!(i, v128_store32_lane::<1>(ab11[i], c![i + 0, 1] as *mut u32));
822        loop4!(i, v128_store32_lane::<2>(ab11[i], c![i + 0, 2] as *mut u32));
823        loop4!(i, v128_store32_lane::<3>(ab11[i], c![i + 0, 3] as *mut u32));
824
825        loop4!(i, v128_store32_lane::<0>(ab12[i], c![i + 0, 4] as *mut u32));
826        loop4!(i, v128_store32_lane::<1>(ab12[i], c![i + 0, 5] as *mut u32));
827        loop4!(i, v128_store32_lane::<2>(ab12[i], c![i + 0, 6] as *mut u32));
828        loop4!(i, v128_store32_lane::<3>(ab12[i], c![i + 0, 7] as *mut u32));
829
830        loop4!(i, v128_store32_lane::<0>(ab21[i], c![i + 4, 0] as *mut u32));
831        loop4!(i, v128_store32_lane::<1>(ab21[i], c![i + 4, 1] as *mut u32));
832        loop4!(i, v128_store32_lane::<2>(ab21[i], c![i + 4, 2] as *mut u32));
833        loop4!(i, v128_store32_lane::<3>(ab21[i], c![i + 4, 3] as *mut u32));
834
835        loop4!(i, v128_store32_lane::<0>(ab22[i], c![i + 4, 4] as *mut u32));
836        loop4!(i, v128_store32_lane::<1>(ab22[i], c![i + 4, 5] as *mut u32));
837        loop4!(i, v128_store32_lane::<2>(ab22[i], c![i + 4, 6] as *mut u32));
838        loop4!(i, v128_store32_lane::<3>(ab22[i], c![i + 4, 7] as *mut u32));
839    }
840}

842#[inline]
843unsafe fn kernel_fallback_impl(k: usize, alpha: T, a: *const T, b: *const T,
844                               beta: T, c: *mut T, rsc: isize, csc: isize)
845{
846    const MR: usize = KernelFallback::MR;
847    const NR: usize = KernelFallback::NR;
848    let mut ab: [[T; NR]; MR] = [[0.; NR]; MR];
849    let mut a = a;
850    let mut b = b;
851    debug_assert_eq!(beta, 0., "Beta must be 0 or is not masked");
852
853    // Compute A B into ab[i][j]
854    unroll_by!(4 => k, {
855        loop8!(i, loop4!(j, ab[i][j] += at(a, i) * at(b, j)));
856
857        a = a.add(MR);
858        b = b.add(NR);
859    });
860
861    macro_rules! c {
862        ($i:expr, $j:expr) => (c.offset(rsc * $i as isize + csc * $j as isize));
863    }
864
865    // set C = α A B
866    loop4!(j, loop8!(i, *c![i, j] = alpha * ab[i][j]));
867}
