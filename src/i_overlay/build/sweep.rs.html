19pub(crate) trait FillHandler<C> {
20    type Output;
21    fn handle(&mut self, index: usize, segment: &Segment<C>, fill: SegmentFill) -> ControlFlow<Self::Output>;
22    fn finalize(self) -> Self::Output;
23}

25#[inline]
26fn sweep_with_handler<C, F, S, H>(scan: &mut S, segments: &[Segment<C>], mut handler: H) -> H::Output
27where
28    C: WindingCount,
29    F: FillStrategy<C>,
30    S: KeyExpCollection<VSegment, i32, C>,
31    H: FillHandler<C>,
32{
33    let mut node = Vec::with_capacity(4);
34    let n = segments.len();
35    let mut i = 0;
36
37    while i < n {
38        let p = segments[i].x_segment.a;
39
40        node.push(End {
41            index: i,
42            point: segments[i].x_segment.b,
43        });
44        i += 1;
45
46        while i < n && segments[i].x_segment.a == p {
47            node.push(End {
48                index: i,
49                point: segments[i].x_segment.b,
50            });
51            i += 1;
52        }
53
54        if node.len() > 1 {
55            node.sort_by(|s0, s1| Triangle::clock_order_point(p, s1.point, s0.point));
56        }
57
58        let mut sum_count = scan.first_less_or_equal_by(p.x, C::new(0, 0), |s| s.is_under_point_order(p));
59
60        for se in node.iter() {
61            let sid = unsafe { segments.get_unchecked(se.index) };
62            let (new_sum, fill) = F::add_and_fill(sid.count, sum_count);
63            sum_count = new_sum;
64
65            if let ControlFlow::Break(result) = handler.handle(se.index, sid, fill) {
66                return result;
67            }
68
69            if sid.x_segment.is_not_vertical() {
70                scan.insert(sid.x_segment.into(), sum_count, p.x);
71            }
72        }
73
74        node.clear();
75    }
76
77    handler.finalize()
78}

80pub(crate) struct SweepRunner<C> {
81    list: Option<KeyExpList<VSegment, i32, C>>,
82    tree: Option<KeyExpTree<VSegment, i32, C>>,
83}
84
85impl<C: WindingCount> SweepRunner<C> {
86    #[inline]
87    pub(crate) fn new() -> Self {
88        Self {
89            list: None,
90            tree: None,
91        }
92    }
93
94    #[inline]
95    pub(crate) fn run<F, H>(&mut self, solver: &Solver, segments: &[Segment<C>], handler: H) -> H::Output
96    where
97        F: FillStrategy<C>,
98        H: FillHandler<C>,
99    {
100        let count = segments.len();
101        if solver.is_list_fill(segments) {
102            let capacity = count.log2_sqrt().max(4) * 2;
103            let mut list = self.take_scan_list(capacity);
104            let result = sweep_with_handler::<C, F, _, _>(&mut list, segments, handler);
105            self.list = Some(list);
106            result
107        } else {
108            let capacity = count.log2_sqrt().max(8);
109            let mut tree = self.take_scan_tree(capacity);
110            let result = sweep_with_handler::<C, F, _, _>(&mut tree, segments, handler);
111            self.tree = Some(tree);
112            result
113        }
114    }
115
116    #[inline]
117    pub(crate) fn run_with_fill_rule<H>(
118        &mut self,
119        fill_rule: FillRule,
120        solver: &Solver,
121        segments: &[Segment<C>],
122        handler: H,
123    ) -> H::Output
124    where
125        H: FillHandler<C>,
126        EvenOddStrategy: FillStrategy<C>,
127        NonZeroStrategy: FillStrategy<C>,
128        PositiveStrategy: FillStrategy<C>,
129        NegativeStrategy: FillStrategy<C>,
130    {
131        match fill_rule {
132            FillRule::EvenOdd => self.run::<EvenOddStrategy, H>(solver, segments, handler),
133            FillRule::NonZero => self.run::<NonZeroStrategy, H>(solver, segments, handler),
134            FillRule::Positive => self.run::<PositiveStrategy, H>(solver, segments, handler),
135            FillRule::Negative => self.run::<NegativeStrategy, H>(solver, segments, handler),
136        }
137    }
138
139    #[inline]
140    fn take_scan_list(&mut self, capacity: usize) -> KeyExpList<VSegment, i32, C> {
141        if let Some(mut list) = self.list.take() {
142            list.clear();
143            list.reserve_capacity(capacity);
144            list
145        } else {
146            KeyExpList::new(capacity)
147        }
148    }
149
150    #[inline]
151    fn take_scan_tree(&mut self, capacity: usize) -> KeyExpTree<VSegment, i32, C> {
152        if let Some(mut tree) = self.tree.take() {
153            tree.clear();
154            tree.reserve_capacity(capacity);
155            tree
156        } else {
157            KeyExpTree::new(capacity)
158        }
159    }
160}
