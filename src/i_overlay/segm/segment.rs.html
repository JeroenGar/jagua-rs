28impl<C: WindingCount> Segment<C> {
29    #[inline(always)]
30    pub(crate) fn create_and_validate(a: IntPoint, b: IntPoint, count: C) -> Self {
31        if a < b {
32            Self {
33                x_segment: XSegment { a, b },
34                count,
35            }
36        } else {
37            Self {
38                x_segment: XSegment { a: b, b: a },
39                count: count.invert(),
40            }
41        }
42    }
43}
