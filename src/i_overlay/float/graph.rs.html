1//! This module defines the graph structure that represents the relationships between the paths in
2//! subject and clip polygons after boolean operations. The graph helps in extracting final shapes
3//! based on the overlay rule applied.

5use crate::core::extract::BooleanExtractionBuffer;
6use crate::core::graph::OverlayGraph;
7use crate::core::overlay_rule::OverlayRule;
8use i_float::adapter::FloatPointAdapter;
9use i_float::float::compatible::FloatPointCompatible;
10use i_float::float::number::FloatNumber;
11use i_shape::base::data::Shapes;
12use i_shape::float::adapter::ShapesToFloat;
13use i_shape::float::despike::DeSpikeContour;
14use i_shape::float::simple::SimplifyContour;

16/// The `FloatOverlayGraph` struct represents an overlay graph with floating point precision,
17/// providing methods to extract geometric shapes from the graph after applying boolean operations.
18/// [More information](https://ishape-rust.github.io/iShape-js/overlay/overlay_graph/overlay_graph.html) about Overlay Graph.
19pub struct FloatOverlayGraph<'a, P: FloatPointCompatible<T>, T: FloatNumber> {
20    pub graph: OverlayGraph<'a>,
21    pub adapter: FloatPointAdapter<P, T>,
22    clean_result: bool,
23}
24
25impl<'a, P: FloatPointCompatible<T>, T: FloatNumber> FloatOverlayGraph<'a, P, T> {
26    #[inline]
27    pub(crate) fn new(graph: OverlayGraph<'a>, adapter: FloatPointAdapter<P, T>, clean_result: bool) -> Self {
28        Self {
29            graph,
30            adapter,
31            clean_result,
32        }
33    }
34
35    /// Extracts shapes from the overlay graph based on the specified overlay rule.
36    /// This method is used to retrieve the final geometric shapes after boolean operations have been applied.
37    /// It's suitable for most use cases where the minimum area of shapes is not a concern.
38    ///
39    /// # Parameters
40    /// - `overlay_rule`: The boolean operation rule to apply when extracting shapes from the graph, such as union or intersection.
41    ///
42    /// # Returns
43    /// A `Shapes<P>` collection, representing the geometric result of the applied overlay rule.
44    ///
45    /// # Shape Representation
46    /// The output is a `Shapes<P>`, where:
47    /// - The outer `Vec<Shape<P>>` represents a set of shapes.
48    /// - Each shape `Vec<Contour<P>>` represents a collection of paths, where the first path is the outer boundary, and all subsequent paths are holes in this boundary.
49    /// - Each path `Vec<P>` is a sequence of points, forming a closed path.
50    ///
51    /// Note: Outer boundary paths have a counterclockwise order, and holes have a clockwise order.
52    #[inline]
53    pub fn extract_shapes(
54        &self,
55        overlay_rule: OverlayRule,
56        buffer: &mut BooleanExtractionBuffer,
57    ) -> Shapes<P> {
58        let shapes = self.graph.extract_shapes(overlay_rule, buffer);
59        let mut float = shapes.to_float(&self.adapter);
60
61        if self.clean_result {
62            if self.graph.options.preserve_output_collinear {
63                float.despike_contour(&self.adapter);
64            } else {
65                float.simplify_contour(&self.adapter);
66            }
67        }
68
69        float
70    }
71}
