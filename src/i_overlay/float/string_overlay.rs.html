14/// The `FloatStringOverlay` struct is a builder for overlaying geometric shapes by converting
15/// floating-point geometry to integer space. It provides methods for adding paths and shapes,
16/// as well as for converting the overlay into a `FloatStringGraph`.
17///
18/// The float-to-integer conversion is controlled by the `FloatPointAdapter` scale:
19/// `x_int = (x_float - offset_x) * scale`. Use a fixed scale if you need predictable precision.
20pub struct FloatStringOverlay<P: FloatPointCompatible<T>, T: FloatNumber> {
21    pub(super) overlay: StringOverlay,
22    pub(super) adapter: FloatPointAdapter<P, T>,
23}
24
25impl<P: FloatPointCompatible<T>, T: FloatNumber> FloatStringOverlay<P, T> {
26    /// Constructs a new `FloatStringOverlay`, a builder for overlaying geometric shapes
27    /// by converting float-based geometry to integer space, using a pre-configured adapter.
28    ///
29    /// - `adapter`: A `FloatPointAdapter` instance responsible for coordinate conversion between
30    ///   float and integer values, ensuring accuracy during geometric transformations.
31    ///   Use `FloatPointAdapter::with_scale` to set a fixed scale, or `FloatPointAdapter::new`
32    ///   for automatic scaling based on bounds.
33    /// - `capacity`: Initial capacity for storing segments, ideally matching the total number of
34    ///   segments for efficient memory allocation.
35    #[inline]
36    pub fn with_adapter(adapter: FloatPointAdapter<P, T>, capacity: usize) -> Self {
37        Self {
38            overlay: StringOverlay::new(capacity),
39            adapter,
40        }
41    }
42
43    /// Creates a new `FloatOverlay` instance and initializes it with subject and clip shapes.
44    ///
45    /// This variant uses automatic scaling based on the combined bounds of `shape` and `string`.
46    /// - `shape`: A `ShapeResource` define the shape.
47    ///   `ShapeResource` can be one of the following:
48    ///     - `Contour`: A contour representing a closed path. This path is interpreted as closed, so it doesn’t require the start and endpoint to be the same for processing.
49    ///     - `Contours`: A collection of contours, each representing a closed path.
50    ///     - `Shapes`: A collection of shapes, where each shape may consist of multiple contours.
51    /// - `string`: A `ShapeResource` define the string paths.
52    ///   `ShapeResource` can be one of the following:
53    ///     - `Path`: A path representing a string line.
54    ///     - `Paths`: A collection of paths, each representing a string line.
55    ///     - `Vec<Paths>`: A collection of grouped paths, where each group may consist of multiple paths.
56    pub fn with_shape_and_string<R0, R1>(shape: &R0, string: &R1) -> Self
57    where
58        R0: ShapeResource<P, T>,
59        R1: ShapeResource<P, T>,
60        P: FloatPointCompatible<T>,
61        T: FloatNumber,
62    {
63        let iter = shape.iter_paths().chain(string.iter_paths()).flatten();
64        let adapter = FloatPointAdapter::with_iter(iter);
65        let shape_capacity = shape.iter_paths().fold(0, |s, c| s + c.len());
66        let string_capacity = string.iter_paths().fold(0, |s, c| s + c.len());
67
68        Self::with_adapter(adapter, shape_capacity + string_capacity)
69            .unsafe_add_shapes(shape)
70            .unsafe_add_string_lines(string)
71    }
72
73    /// Creates a new `FloatStringOverlay` instance with a fixed float-to-integer scale.
74    ///
75    /// This variant validates that the requested scale is finite, positive, and fits the
76    /// input bounds. Use `scale = 1.0 / grid_size` if you want a grid-size style parameter.
77    pub fn with_shape_and_string_fixed_scale<R0, R1>(
78        shape: &R0,
79        string: &R1,
80        scale: T,
81    ) -> Result<Self, FixedScaleOverlayError>
82    where
83        R0: ShapeResource<P, T>,
84        R1: ShapeResource<P, T>,
85        P: FloatPointCompatible<T>,
86        T: FloatNumber,
87    {
88        let s = FixedScaleOverlayError::validate_scale(scale)?;
89
90        let iter = shape.iter_paths().chain(string.iter_paths()).flatten();
91        let mut adapter = FloatPointAdapter::with_iter(iter);
92        if adapter.dir_scale < scale {
93            return Err(FixedScaleOverlayError::ScaleTooLarge);
94        }
95
96        adapter.dir_scale = scale;
97        adapter.inv_scale = T::from_float(1.0 / s);
98
99        let shape_capacity = shape.iter_paths().fold(0, |s, c| s + c.len());
100        let string_capacity = string.iter_paths().fold(0, |s, c| s + c.len());
101
102        Ok(Self::with_adapter(adapter, shape_capacity + string_capacity)
103            .unsafe_add_shapes(shape)
104            .unsafe_add_string_lines(string))
105    }

107    /// Adds a shapes to the overlay.
108    /// - `source`: A `ShapeResource` that define shape.
109    ///   `ShapeResource` can be one of the following:
110    ///     - `Contour`: A contour representing a closed path. This path is interpreted as closed, so it doesn’t require the start and endpoint to be the same for processing.
111    ///     - `Contours`: A collection of contours, each representing a closed path.
112    ///     - `Shapes`: A collection of shapes, where each shape may consist of multiple contours.
113    /// - `shape_type`: Specifies the role of the added paths in the overlay operation, either as `Subject` or `Clip`.
114    #[inline]
115    pub fn unsafe_add_shapes<S: ShapeResource<P, T>>(mut self, source: &S) -> Self {
116        for contour in source.iter_paths() {
117            self = self.unsafe_add_shape_contour(contour);
118        }
119        self
120    }
121
122    /// Adds a string line paths to the overlay.
123    /// - `resource`: A `ShapeResource` that define shape.
124    ///   `ShapeResource` can be one of the following:
125    ///     - `Path`: A path representing a string line.
126    ///     - `Paths`: A collection of paths, each representing a string line.
127    ///     - `Vec<Paths>`: A collection of grouped paths, where each group may consist of multiple paths.
128    #[inline]
129    pub fn unsafe_add_string_lines<S: ShapeResource<P, T>>(mut self, resource: &S) -> Self {
130        for path in resource.iter_paths() {
131            self = self.unsafe_add_string_line(path);
132        }
133        self
134    }
135
136    /// Adds a closed shape path to the overlay.
137    /// - `contour`: An array of points that form a closed path.
138    /// - **Safety**: Marked `unsafe` because it assumes the path is fully contained within the bounding box.
139    #[inline]
140    pub fn unsafe_add_shape_contour(mut self, contour: &[P]) -> Self {
141        self.overlay
142            .add_shape_contour_iter(contour.iter().map(|p| self.adapter.float_to_int(p)));
143        self
144    }
145
146    /// Adds an open string line path to the overlay.
147    /// - `path`: A path representing a string line.
148    /// - **Safety**: Marked `unsafe` because it assumes each path is fully contained within the bounding box.
149    #[inline]
150    pub fn unsafe_add_string_line(mut self, path: &[P]) -> Self {
151        for w in path.windows(2) {
152            let a = self.adapter.float_to_int(&w[0]);
153            let b = self.adapter.float_to_int(&w[1]);
154            self.overlay.add_string_line([a, b]);
155        }
156
157        self
158    }

160    /// Converts the current overlay into an `FloatStringGraph` based on the specified build rule.
161    /// The resulting graph is the foundation for performing boolean operations, and it's optimized for such operations based on the provided build rule.
162    /// - `fill_rule`: Fill rule to determine filled areas (non-zero, even-odd, positive, negative).
163    /// - Returns: A `FloatStringGraph` containing the graph representation of the overlay's geometry.
164    #[inline]
165    pub fn build_graph_view(&mut self, fill_rule: FillRule) -> Option<FloatStringGraph<'_, P, T>> {
166        self.build_graph_view_with_solver(fill_rule, Default::default())
167    }
168
169    /// Converts the current overlay into an `FloatStringGraph` based on the specified build rule and solver.
170    /// This method allows for finer control over the boolean operation process by passing a custom solver.
171    /// - `fill_rule`: Fill rule to determine filled areas (non-zero, even-odd, positive, negative).
172    /// - `solver`: A custom solver for optimizing or modifying the graph creation process.
173    /// - Returns: A `FloatStringGraph` containing the graph representation of the overlay's geometry.
174    #[inline]
175    pub fn build_graph_view_with_solver(
176        &mut self,
177        fill_rule: FillRule,
178        solver: Solver,
179    ) -> Option<FloatStringGraph<'_, P, T>> {
180        let graph = self.overlay.build_graph_view_with_solver(fill_rule, solver)?;
181        Some(FloatStringGraph {
182            graph,
183            adapter: self.adapter.clone(),
184        })
185    }
186
187    /// Executes a single Boolean operation on the current geometry using the specified build and clip rules.
188    ///
189    /// ### Parameters:
190    /// - `fill_rule`: Fill rule to determine filled areas (non-zero, even-odd, positive, negative).
191    /// - `clip_rule`: Clip rule to determine how the boundary and inversion settings affect the result.
192    /// - `solver`: Type of solver to use.
193    /// - Returns: A `Paths<P>` collection of string lines that meet the clipping conditions.
194    #[inline]
195    pub fn clip_string_lines_with_solver(
196        self,
197        fill_rule: FillRule,
198        clip_rule: ClipRule,
199        solver: Solver,
200    ) -> Paths<P> {
201        let paths = self
202            .overlay
203            .clip_string_lines_with_solver(fill_rule, clip_rule, solver);
204        paths.to_float(&self.adapter)
205    }
206}

213    #[test]
214    fn test_fixed_scale_ok() {
215        let shape = vec![vec![[0.0, 0.0], [0.0, 1.0], [1.0, 1.0], [1.0, 0.0]]];
216        let string = vec![[0.0, 0.5], [2.0, 0.5]];
217
218        let overlay = FloatStringOverlay::with_shape_and_string_fixed_scale(&shape, &string, 10.0);
219
220        assert!(overlay.is_ok());
221    }
222
223    #[test]
224    fn test_fixed_scale_invalid() {
225        let shape = vec![vec![[0.0, 0.0], [0.0, 1.0], [1.0, 1.0], [1.0, 0.0]]];
226        let string = vec![[0.0, 0.5], [2.0, 0.5]];
227
228        assert!(FloatStringOverlay::with_shape_and_string_fixed_scale(&shape, &string, 0.0).is_err());
229        assert!(FloatStringOverlay::with_shape_and_string_fixed_scale(&shape, &string, -1.0).is_err());
230        assert!(FloatStringOverlay::with_shape_and_string_fixed_scale(&shape, &string, f64::NAN).is_err());
231        assert!(
232            FloatStringOverlay::with_shape_and_string_fixed_scale(&shape, &string, f64::INFINITY).is_err()
233        );
234    }
