25pub trait OutlineOffset<P: FloatPointCompatible<T>, T: FloatNumber> {
26    /// Generates an outline shapes for contours, or shapes.
27    ///
28    /// - `style`: Defines the outline properties, including offset, and joins.
29    ///
30    /// # Returns
31    /// A collection of `Shapes<P>` representing the outline geometry.
32    /// Note: Outer boundary paths have a counterclockwise order, and holes have a clockwise order.
33    fn outline(&self, style: &OutlineStyle<T>) -> Shapes<P>;
34
35    /// Generates outline contours directly into a flat buffer.
36    ///
37    /// - `style`: Defines the outline properties, including offset, and joins.
38    /// - `output`: Destination buffer that receives resulting contours. Existing contents are replaced.
39    fn outline_into(&self, style: &OutlineStyle<T>, output: &mut FloatFlatContoursBuffer<P>);
40
41    /// Generates an outline shapes for contours, or shapes with optional filtering.
42    ///
43    /// - `style`: Defines the outline properties, including offset, and joins.
44    /// - `options`: Adjust custom behavior.
45    ///
46    /// # Returns
47    /// A collection of `Shapes<P>` representing the outline geometry.
48    /// Note: Outer boundary paths have a **main_direction** order, and holes have an opposite to **main_direction** order.
49    fn outline_custom(&self, style: &OutlineStyle<T>, options: OverlayOptions<T>) -> Shapes<P>;
50
51    /// Generates outline contours directly into a flat buffer with optional filtering.
52    ///
53    /// - `style`: Defines the outline properties, including offset, and joins.
54    /// - `options`: Adjust custom behavior.
55    /// - `output`: Destination buffer that receives resulting contours. Existing contents are replaced.
56    fn outline_custom_into(
57        &self,
58        style: &OutlineStyle<T>,
59        options: OverlayOptions<T>,
60        output: &mut FloatFlatContoursBuffer<P>,
61    );
62
63    /// Generates an outline shapes for contours, or shapes with a fixed float-to-integer scale.
64    ///
65    /// - `style`: Defines the outline properties, including offset, and joins.
66    /// - `scale`: Fixed float-to-integer scale. Use `scale = 1.0 / grid_size` if you prefer grid size semantics.
67    ///
68    /// # Returns
69    /// A collection of `Shapes<P>` representing the outline geometry.
70    /// Note: Outer boundary paths have a counterclockwise order, and holes have a clockwise order.
71    fn outline_fixed_scale(
72        &self,
73        style: &OutlineStyle<T>,
74        scale: T,
75    ) -> Result<Shapes<P>, FixedScaleOverlayError>;
76
77    /// Generates outline contours directly into a flat buffer with fixed float-to-integer scale.
78    ///
79    /// - `style`: Defines the outline properties, including offset, and joins.
80    /// - `scale`: Fixed float-to-integer scale. Use `scale = 1.0 / grid_size` if you prefer grid size semantics.
81    /// - `output`: Destination buffer that receives resulting contours. Existing contents are replaced on success.
82    fn outline_fixed_scale_into(
83        &self,
84        style: &OutlineStyle<T>,
85        scale: T,
86        output: &mut FloatFlatContoursBuffer<P>,
87    ) -> Result<(), FixedScaleOverlayError>;
88
89    /// Generates an outline shapes for contours, or shapes with optional filtering and fixed scaling.
90    ///
91    /// - `style`: Defines the outline properties, including offset, and joins.
92    /// - `options`: Adjust custom behavior.
93    /// - `scale`: Fixed float-to-integer scale. Use `scale = 1.0 / grid_size` if you prefer grid size semantics.
94    ///
95    /// # Returns
96    /// A collection of `Shapes<P>` representing the outline geometry.
97    /// Note: Outer boundary paths have a **main_direction** order, and holes have an opposite to **main_direction** order.
98    fn outline_custom_fixed_scale(
99        &self,
100        style: &OutlineStyle<T>,
101        options: OverlayOptions<T>,
102        scale: T,
103    ) -> Result<Shapes<P>, FixedScaleOverlayError>;
104
105    /// Generates outline contours directly into a flat buffer with optional filtering and fixed scaling.
106    ///
107    /// - `style`: Defines the outline properties, including offset, and joins.
108    /// - `options`: Adjust custom behavior.
109    /// - `scale`: Fixed float-to-integer scale. Use `scale = 1.0 / grid_size` if you prefer grid size semantics.
110    /// - `output`: Destination buffer that receives resulting contours. Existing contents are replaced on success.
111    fn outline_custom_fixed_scale_into(
112        &self,
113        style: &OutlineStyle<T>,
114        options: OverlayOptions<T>,
115        scale: T,
116        output: &mut FloatFlatContoursBuffer<P>,
117    ) -> Result<(), FixedScaleOverlayError>;
118}

120impl<S, P, T> OutlineOffset<P, T> for S
121where
122    S: ShapeResource<P, T>,
123    P: FloatPointCompatible<T> + 'static,
124    T: FloatNumber + 'static,
125{
126    fn outline(&self, style: &OutlineStyle<T>) -> Shapes<P> {
127        self.outline_custom(style, Default::default())
128    }
129
130    fn outline_into(&self, style: &OutlineStyle<T>, output: &mut FloatFlatContoursBuffer<P>) {
131        self.outline_custom_into(style, Default::default(), output)
132    }
133
134    fn outline_custom(&self, style: &OutlineStyle<T>, options: OverlayOptions<T>) -> Shapes<P> {
135        match OutlineSolver::prepare(self, style) {
136            Some(solver) => solver.build(self, options),
137            None => vec![],
138        }
139    }
140
141    fn outline_custom_into(
142        &self,
143        style: &OutlineStyle<T>,
144        options: OverlayOptions<T>,
145        output: &mut FloatFlatContoursBuffer<P>,
146    ) {
147        match OutlineSolver::prepare(self, style) {
148            Some(solver) => solver.build_into(self, options, output),
149            None => output.clear_and_reserve(0, 0),
150        }
151    }
152
153    fn outline_fixed_scale(
154        &self,
155        style: &OutlineStyle<T>,
156        scale: T,
157    ) -> Result<Shapes<P>, FixedScaleOverlayError> {
158        self.outline_custom_fixed_scale(style, Default::default(), scale)
159    }
160
161    fn outline_fixed_scale_into(
162        &self,
163        style: &OutlineStyle<T>,
164        scale: T,
165        output: &mut FloatFlatContoursBuffer<P>,
166    ) -> Result<(), FixedScaleOverlayError> {
167        self.outline_custom_fixed_scale_into(style, Default::default(), scale, output)
168    }
169
170    fn outline_custom_fixed_scale(
171        &self,
172        style: &OutlineStyle<T>,
173        options: OverlayOptions<T>,
174        scale: T,
175    ) -> Result<Shapes<P>, FixedScaleOverlayError> {
176        let s = FixedScaleOverlayError::validate_scale(scale)?;
177        let mut solver = match OutlineSolver::prepare(self, style) {
178            Some(solver) => solver,
179            None => return Ok(vec![]),
180        };
181        solver.apply_scale(s)?;
182        Ok(solver.build(self, options))
183    }
184
185    fn outline_custom_fixed_scale_into(
186        &self,
187        style: &OutlineStyle<T>,
188        options: OverlayOptions<T>,
189        scale: T,
190        output: &mut FloatFlatContoursBuffer<P>,
191    ) -> Result<(), FixedScaleOverlayError> {
192        let s = FixedScaleOverlayError::validate_scale(scale)?;
193        let mut solver = match OutlineSolver::prepare(self, style) {
194            Some(solver) => solver,
195            None => {
196                output.clear_and_reserve(0, 0);
197                return Ok(());
198            }
199        };
200        solver.apply_scale(s)?;
201        solver.build_into(self, options, output);
202        Ok(())
203    }
204}

206struct OutlineSolver<P: FloatPointCompatible<T>, T: FloatNumber> {
207    outer_builder: OutlineBuilder<P, T>,
208    inner_builder: OutlineBuilder<P, T>,
209    adapter: FloatPointAdapter<P, T>,
210    points_count: usize,
211}
212
213impl<P: FloatPointCompatible<T> + 'static, T: FloatNumber + 'static> OutlineSolver<P, T> {
214    fn prepare<S: ShapeResource<P, T>>(source: &S, style: &OutlineStyle<T>) -> Option<Self> {
215        let (points_count, paths_count) = {
216            let mut points_count = 0;
217            let mut paths_count = 0;
218            for path in source.iter_paths() {
219                points_count += path.len();
220                paths_count += 1;
221            }
222            (points_count, paths_count)
223        };
224
225        if paths_count == 0 {
226            return None;
227        }
228
229        let join = style.join.clone().normalize();
230        let outer_builder = OutlineBuilder::new(-style.outer_offset, &join);
231        let inner_builder = OutlineBuilder::new(-style.inner_offset, &join);
232
233        let outer_radius = style.outer_offset;
234        let inner_radius = style.inner_offset;
235
236        let outer_additional_offset = outer_builder.additional_offset(outer_radius);
237        let inner_additional_offset = inner_builder.additional_offset(inner_radius);
238
239        let additional_offset = outer_additional_offset.abs() + inner_additional_offset.abs();
240
241        let mut rect = FloatRect::with_iter(source.iter_paths().flatten()).unwrap_or(FloatRect::zero());
242        rect.add_offset(additional_offset);
243
244        let adapter = FloatPointAdapter::new(rect);
245
246        Some(Self {
247            outer_builder,
248            inner_builder,
249            adapter,
250            points_count,
251        })
252    }
253
254    fn apply_scale(&mut self, scale: f64) -> Result<(), FixedScaleOverlayError> {
255        let s = T::from_float(scale);
256        if self.adapter.dir_scale < s {
257            return Err(FixedScaleOverlayError::ScaleTooLarge);
258        }
259
260        self.adapter.dir_scale = s;
261        self.adapter.inv_scale = T::from_float(1.0 / scale);
262
263        Ok(())
264    }
265
266    fn build_overlay<S: ShapeResource<P, T>>(&self, source: &S, options: OverlayOptions<T>) -> Overlay {
267        let total_capacity = self.outer_builder.capacity(self.points_count);
268        let mut overlay = Overlay::new_custom(
269            total_capacity,
270            options.int_with_adapter(&self.adapter),
271            Default::default(),
272        );
273
274        let mut offset_overlay = Overlay::new(16);
275        offset_overlay.options = overlay.options;
276
277        let mut segments = Vec::new();
278        let mut bool_buffer = BooleanExtractionBuffer::default();
279        let mut flat_buffer = FlatContoursBuffer::default();
280
281        for path in source.iter_paths() {
282            let area = path.unsafe_int_area(&self.adapter);
283            if area.abs() <= 1 {
284                // ignore degenerate paths
285                continue;
286            }
287
288            offset_overlay.clear();
289            segments.clear();
290
291            if area < 0 {
292                offset_overlay.options.output_direction = ContourDirection::CounterClockwise;
293                segments.reserve(self.outer_builder.capacity(path.len()));
294                self.outer_builder.build(path, &self.adapter, &mut segments);
295
296                offset_overlay.add_segments(&segments);
297
298                if let Some(graph) = offset_overlay.build_graph_view(FillRule::Positive) {
299                    graph.extract_contours_into(OverlayRule::Subject, &mut bool_buffer, &mut flat_buffer);
300                }
301            } else {
302                offset_overlay.options.output_direction = ContourDirection::Clockwise;
303                segments.reserve(self.inner_builder.capacity(path.len()));
304                self.inner_builder.build(path, &self.adapter, &mut segments);
305
306                offset_overlay.add_segments(&segments);
307
308                if let Some(graph) = offset_overlay.build_graph_view(FillRule::Negative) {
309                    graph.extract_contours_into(OverlayRule::Subject, &mut bool_buffer, &mut flat_buffer);
310                }
311            }
312
313            overlay.add_flat_buffer(&flat_buffer, Subject);
314        }
315
316        overlay
317    }
318
319    fn build<S: ShapeResource<P, T>>(self, source: &S, options: OverlayOptions<T>) -> Shapes<P> {
320        let preserve_output_collinear = options.preserve_output_collinear;
321        let clean_result = options.clean_result;
322        let mut overlay = self.build_overlay(source, options);
323        let shapes = overlay.overlay(OverlayRule::Subject, FillRule::Positive);
324
325        if clean_result {
326            let mut float = shapes.to_float(&self.adapter);
327            if preserve_output_collinear {
328                float.despike_contour(&self.adapter);
329            } else {
330                float.simplify_contour(&self.adapter);
331            }
332            float
333        } else {
334            shapes.to_float(&self.adapter)
335        }
336    }
337
338    fn build_into<S: ShapeResource<P, T>>(
339        self,
340        source: &S,
341        options: OverlayOptions<T>,
342        output: &mut FloatFlatContoursBuffer<P>,
343    ) {
344        let preserve_output_collinear = options.preserve_output_collinear;
345        let clean_result = options.clean_result;
346        let mut overlay = self.build_overlay(source, options);
347
348        let mut int_output = FlatContoursBuffer::default();
349        overlay.overlay_into(OverlayRule::Subject, FillRule::Positive, &mut int_output);
350        let iter = int_output.points.iter().map(|p| self.adapter.int_to_float(p));
351        output.set_with_iter(iter, &int_output.ranges);
352
353        if clean_result {
354            if preserve_output_collinear {
355                output.despike_contour(&self.adapter);
356            } else {
357                output.simplify_contour(&self.adapter);
358            }
359        }
360    }
361}
