16/// Overlay structure optimized for spatial predicate evaluation.
17///
18/// `PredicateOverlay` provides efficient spatial relationship testing between
19/// two polygon sets without computing full boolean operation results. It is
20/// designed for cases where you only need to know *whether* shapes intersect,
21/// not *what* the intersection looks like.
22///
23/// # Example
24///
25/// ```
26/// use i_overlay::core::relate::PredicateOverlay;
27/// use i_overlay::core::overlay::ShapeType;
28///
29/// let mut overlay = PredicateOverlay::new(16);
30/// // Add subject and clip segments...
31/// let intersects = overlay.intersects();
32/// ```
33///
34/// For float coordinates, prefer using [`FloatPredicateOverlay`](crate::float::relate::FloatPredicateOverlay)
35/// or the [`FloatRelate`](crate::float::relate::FloatRelate) trait.
36pub struct PredicateOverlay {
37    /// Solver configuration for segment operations.
38    pub solver: Solver,
39    /// Fill rule for determining polygon interiors.
40    pub fill_rule: FillRule,
41    pub(crate) segments: Vec<Segment<ShapeCountBoolean>>,
42    pub(crate) split_solver: SplitSolver,
43    sweep_runner: SweepRunner<ShapeCountBoolean>,
44}
45
46impl PredicateOverlay {
47    #[inline]
48    pub fn new(capacity: usize) -> Self {
49        Self {
50            solver: Default::default(),
51            fill_rule: FillRule::EvenOdd,
52            segments: Vec::with_capacity(capacity),
53            split_solver: SplitSolver::new(),
54            sweep_runner: SweepRunner::new(),
55        }
56    }
57
58    fn evaluate<T: Default, H: FillHandler<ShapeCountBoolean, Output = T>>(&mut self, handler: H) -> T {
59        if self.segments.is_empty() {
60            return T::default();
61        }
62        self.split_solver.split_segments(&mut self.segments, &self.solver);
63        if self.segments.is_empty() {
64            return T::default();
65        }
66        self.sweep_runner
67            .run_with_fill_rule(self.fill_rule, &self.solver, &self.segments, handler)
68    }
69
70    /// Returns `true` if the subject and clip shapes intersect (share any point).
71    ///
72    /// This includes both interior overlap and boundary contact (including single-point touches).
73    #[inline]
74    pub fn intersects(&mut self) -> bool {
75        let capacity = self.segments.len();
76        self.evaluate(IntersectsHandler::new(capacity))
77    }
78
79    /// Returns `true` if the interiors of subject and clip shapes overlap.
80    ///
81    /// Unlike `intersects()`, this returns `false` for shapes that only share
82    /// boundary points (edges or vertices) without interior overlap.
83    #[inline]
84    pub fn interiors_intersect(&mut self) -> bool {
85        self.evaluate(InteriorsIntersectHandler)
86    }
87
88    /// Returns `true` if subject and clip shapes touch (boundaries intersect but interiors don't).
89    ///
90    /// This returns `true` when shapes share boundary points (edges or vertices)
91    /// but their interiors don't overlap. This includes single-point vertex touches.
92    #[inline]
93    pub fn touches(&mut self) -> bool {
94        let capacity = self.segments.len();
95        self.evaluate(TouchesHandler::new(capacity))
96    }
97
98    /// Returns `true` if subject and clip shapes intersect by point coincidence only.
99    ///
100    /// This returns `true` when shapes share boundary vertices but NOT edges.
101    /// Unlike `touches()`, this returns `false` for shapes that share edges.
102    #[inline]
103    pub fn point_intersects(&mut self) -> bool {
104        let capacity = self.segments.len();
105        self.evaluate(PointIntersectsHandler::new(capacity))
106    }
107
108    /// Returns `true` if subject is completely within clip.
109    ///
110    /// Subject is within clip if everywhere the subject has fill, the clip
111    /// also has fill on the same side.
112    #[inline]
113    pub fn within(&mut self) -> bool {
114        self.evaluate(WithinHandler::new())
115    }
116
117    /// Adds a path to the overlay using an iterator, allowing for more flexible path input.
118    /// This function is particularly useful when working with dynamically generated paths or
119    /// when paths are not directly stored in a collection.
120    /// - `iter`: An iterator over references to `IntPoint` that defines the path.
121    /// - `shape_type`: Specifies the role of the added path in the overlay operation, either as `Subject` or `Clip`.
122    #[inline]
123    pub fn add_path_iter<I: Iterator<Item = IntPoint>>(&mut self, iter: I, shape_type: ShapeType) {
124        self.segments.append_path_iter(iter, shape_type, false);
125    }
126
127    /// Adds a single path to the overlay as either subject or clip paths.
128    /// - `contour`: An array of points that form a closed path.
129    /// - `shape_type`: Specifies the role of the added path in the overlay operation, either as `Subject` or `Clip`.
130    #[inline]
131    pub fn add_contour(&mut self, contour: &[IntPoint], shape_type: ShapeType) {
132        self.segments
133            .append_path_iter(contour.iter().copied(), shape_type, false);
134    }
135
136    /// Adds multiple paths to the overlay as either subject or clip paths.
137    /// - `contours`: An array of `IntContour` instances to be added to the overlay.
138    /// - `shape_type`: Specifies the role of the added paths in the overlay operation, either as `Subject` or `Clip`.
139    #[inline]
140    pub fn add_contours(&mut self, contours: &[IntContour], shape_type: ShapeType) {
141        for contour in contours.iter() {
142            self.add_contour(contour, shape_type);
143        }
144    }
145
146    /// Adds a single shape to the overlay as either a subject or clip shape.
147    /// - `shape`: A reference to a `IntShape` instance to be added.
148    /// - `shape_type`: Specifies the role of the added shape in the overlay operation, either as `Subject` or `Clip`.
149    #[inline]
150    pub fn add_shape(&mut self, shape: &IntShape, shape_type: ShapeType) {
151        self.add_contours(shape, shape_type);
152    }
153
154    /// Adds multiple shapes to the overlay as either subject or clip shapes.
155    /// - `shapes`: An array of `IntShape` instances to be added to the overlay.
156    /// - `shape_type`: Specifies the role of the added shapes in the overlay operation, either as `Subject` or `Clip`.
157    #[inline]
158    pub fn add_shapes(&mut self, shapes: &[IntShape], shape_type: ShapeType) {
159        for shape in shapes.iter() {
160            self.add_contours(shape, shape_type);
161        }
162    }
163
164    #[inline]
165    pub fn clear(&mut self) {
166        self.segments.clear();
167    }
168}

287    /// Test that intersects() detects when an edge passes through another polygon's interior.
288    ///
289    /// Shape A is a quadrilateral: (0,1), (0,0), (3,0), (3,8)
290    /// Shape B is a box: (0,3) to (1,4)
291    ///
292    /// The edge from (0,1) to (3,8) passes through (1, 10/3 ≈ 3.33), which is inside B.
293    /// Therefore intersects() should return true.
294    #[test]
295    fn test_intersects_edge_through_interior() {
296        // Shape A: quadrilateral with edge passing through B's interior
297        let shape_a = vec![
298            IntPoint::new(0, 1),
299            IntPoint::new(0, 0),
300            IntPoint::new(3, 0),
301            IntPoint::new(3, 8),
302        ];
303
304        // Shape B: box from (0,3) to (1,4)
305        let shape_b = vec![
306            IntPoint::new(0, 3),
307            IntPoint::new(1, 3),
308            IntPoint::new(1, 4),
309            IntPoint::new(0, 4),
310        ];
311
312        let mut overlay = PredicateOverlay::new(16);
313        overlay.add_contour(&shape_a, ShapeType::Subject);
314        overlay.add_contour(&shape_b, ShapeType::Clip);
315
316        // The edge from (0,1) to (3,8) has parametric form:
317        // P(t) = (0,1) + t*(3,7) = (3t, 1+7t) for t in [0,1]
318        // At x=1: t=1/3, y = 1 + 7/3 = 10/3 ≈ 3.333
319        // Point (1, 3.333) is strictly inside B (x in [0,1], y in [3,4])
320        assert!(
321            overlay.intersects(),
322            "Edge (0,1)->(3,8) passes through box interior at (1, 3.33); should intersect"
323        );
324    }

357    /// Creates a square with a hole (doughnut shape).
358    /// Outer: counter-clockwise, Inner hole: clockwise
359    fn doughnut(
360        outer_x: i32,
361        outer_y: i32,
362        outer_size: i32,
363        hole_x: i32,
364        hole_y: i32,
365        hole_size: i32,
366    ) -> Vec<Vec<IntPoint>> {
367        vec![
368            // Outer boundary (counter-clockwise)
369            vec![
370                IntPoint::new(outer_x, outer_y),
371                IntPoint::new(outer_x, outer_y + outer_size),
372                IntPoint::new(outer_x + outer_size, outer_y + outer_size),
373                IntPoint::new(outer_x + outer_size, outer_y),
374            ],
375            // Inner hole (clockwise)
376            vec![
377                IntPoint::new(hole_x, hole_y),
378                IntPoint::new(hole_x + hole_size, hole_y),
379                IntPoint::new(hole_x + hole_size, hole_y + hole_size),
380                IntPoint::new(hole_x, hole_y + hole_size),
381            ],
382        ]
383    }

385    /// Creates a diamond (rotated square) with corners at midpoints of a bounding box.
386    fn diamond(cx: i32, cy: i32, radius: i32) -> Vec<IntPoint> {
387        vec![
388            IntPoint::new(cx, cy - radius), // top
389            IntPoint::new(cx + radius, cy), // right
390            IntPoint::new(cx, cy + radius), // bottom
391            IntPoint::new(cx - radius, cy), // left
392        ]
393    }
