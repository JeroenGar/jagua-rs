37impl OverlayGraph<'_> {
38    /// Extracts shapes from the overlay graph based on the specified overlay rule. This method is used to retrieve the final geometric shapes after boolean operations have been applied. It's suitable for most use cases where the minimum area of shapes is not a concern.
39    /// - `overlay_rule`: The boolean operation rule to apply when extracting shapes from the graph, such as union or intersection.
40    /// - `buffer`: Reusable buffer, optimisation purpose only.
41    /// - Returns: A vector of `IntShape`, representing the geometric result of the applied overlay rule.
42    /// # Shape Representation
43    /// The output is a `IntShapes`, where:
44    /// - The outer `Vec<IntShape>` represents a set of shapes.
45    /// - Each shape `Vec<IntContour>` represents a collection of contours, where the first contour is the outer boundary, and all subsequent contours are holes in this boundary.
46    /// - Each path `Vec<IntPoint>` is a sequence of points, forming a closed path.
47    ///
48    /// Note: Outer boundary paths have a counterclockwise order, and holes have a clockwise order.
49    #[inline]
50    pub fn extract_shapes(
51        &self,
52        overlay_rule: OverlayRule,
53        buffer: &mut BooleanExtractionBuffer,
54    ) -> IntShapes {
55        self.links
56            .filter_by_overlay_into(overlay_rule, &mut buffer.visited);
57        if self.options.ogc {
58            self.extract_ogc(overlay_rule, buffer)
59        } else {
60            self.extract(overlay_rule, buffer)
61        }
62    }

64    /// Extracts the flat contours from the overlay graph based on the specified overlay rule.
65    ///
66    /// This method performs a Boolean operation (e.g., union or intersection) and stores the result
67    /// directly into a flat buffer of contours, without nesting them into shapes (i.e., no hole-joining or grouping).
68    ///
69    /// It is optimized for performance and suitable when raw contour data is sufficient,
70    /// such as during intermediate processing, visualization, or tesselation.
71    ///
72    /// - `overlay_rule`: The boolean operation rule to apply (e.g., union, intersection, xor).
73    /// - `buffer`: Reusable working buffer to avoid reallocations.
74    /// - `output`: A flat buffer to which the resulting valid contours will be written.
75    #[inline]
76    pub fn extract_contours_into(
77        &self,
78        overlay_rule: OverlayRule,
79        buffer: &mut BooleanExtractionBuffer,
80        output: &mut FlatContoursBuffer,
81    ) {
82        self.links
83            .filter_by_overlay_into(overlay_rule, &mut buffer.visited);
84        self.extract_contours(overlay_rule, buffer, output);
85    }

87    pub(crate) fn extract(
88        &self,
89        overlay_rule: OverlayRule,
90        buffer: &mut BooleanExtractionBuffer,
91    ) -> IntShapes {
92        let clockwise = self.options.output_direction == ContourDirection::Clockwise;
93
94        let mut shapes = Vec::new();
95        let mut holes = Vec::new();
96        let mut anchors = Vec::new();
97
98        buffer.points.reserve_capacity(buffer.visited.len());
99
100        let mut link_index = 0;
101        let mut anchors_already_sorted = true;
102        while link_index < buffer.visited.len() {
103            if buffer.visited.is_visited(link_index) {
104                link_index += 1;
105                continue;
106            }
107
108            let left_top_link = unsafe {
109                // Safety: `link_index` walks 0..buffer.visited.len(), and buffer.visited.len() <= self.links.len().
110                GraphUtil::find_left_top_link(self.links, self.nodes, link_index, &buffer.visited)
111            };
112
113            let link = unsafe {
114                // Safety: `left_top_link` originates from `find_left_top_link`, which only returns
115                // indices in 0..self.links.len(), so this lookup cannot go out of bounds.
116                self.links.get_unchecked(left_top_link)
117            };
118            let is_hole = overlay_rule.is_fill_top(link.fill);
119            let visited_state = [VisitState::HullVisited, VisitState::HoleVisited][is_hole as usize];
120
121            let direction = is_hole == clockwise;
122            let start_data = StartPathData::new(direction, link, left_top_link);
123
124            self.find_contour(
125                &start_data,
126                direction,
127                visited_state,
128                &mut buffer.visited,
129                &mut buffer.points,
130            );
131            let (is_valid, is_modified) = buffer.points.validate(
132                self.options.min_output_area,
133                self.options.preserve_output_collinear,
134            );
135
136            if !is_valid {
137                link_index += 1;
138                continue;
139            }
140
141            let contour = buffer.points.as_slice().to_vec();
142
143            if is_hole {
144                let mut v_segment = if clockwise {
145                    VSegment {
146                        a: contour[1],
147                        b: contour[2],
148                    }
149                } else {
150                    VSegment {
151                        a: contour[0],
152                        b: contour[contour.len() - 1],
153                    }
154                };
155                if is_modified {
156                    let most_left = contour.left_bottom_segment();
157                    if most_left != v_segment {
158                        v_segment = most_left;
159                        anchors_already_sorted = false;
160                    }
161                };
162
163                debug_assert_eq!(v_segment, contour.left_bottom_segment());
164                let id_data = ContourIndex::new_hole(holes.len());
165                anchors.push(IdSegment::with_segment(id_data, v_segment));
166                holes.push(contour);
167            } else {
168                shapes.push(vec![contour]);
169            }
170        }
171
172        if !anchors_already_sorted {
173            anchors.sort_by(|s0, s1| s0.v_segment.a.cmp(&s1.v_segment.a));
174        }
175
176        shapes.join_sorted_holes(holes, anchors, clockwise);
177
178        shapes
179    }

181    pub(crate) fn find_contour(
182        &self,
183        start_data: &StartPathData,
184        clockwise: bool,
185        visited_state: VisitState,
186        visited: &mut [VisitState],
187        points: &mut Vec<IntPoint>,
188    ) {
189        let mut link_id = start_data.link_id;
190        let mut node_id = start_data.node_id;
191        let last_node_id = start_data.last_node_id;
192
193        visited.visit_edge(link_id, visited_state);
194        points.clear();
195        points.push(start_data.begin);
196
197        // Find a closed tour
198        while node_id != last_node_id {
199            link_id = GraphUtil::next_link(self.links, self.nodes, link_id, node_id, clockwise, visited);
200
201            let link = unsafe {
202                // Safety: `link_id` is always derived from a previous in-bounds index or
203                // from `find_left_top_link`, so it remains in `0..self.links.len()`.
204                self.links.get_unchecked(link_id)
205            };
206            node_id = points.push_node_and_get_other(link, node_id);
207
208            visited.visit_edge(link_id, visited_state);
209        }
210    }

212    fn extract_contours(
213        &self,
214        overlay_rule: OverlayRule,
215        buffer: &mut BooleanExtractionBuffer,
216        output: &mut FlatContoursBuffer,
217    ) {
218        let clockwise = self.options.output_direction == ContourDirection::Clockwise;
219        let len = buffer.visited.len();
220        buffer.points.reserve_capacity(len);
221        output.clear_and_reserve(len, 4);
222
223        let mut link_index = 0;
224        while link_index < len {
225            if buffer.visited.is_visited(link_index) {
226                link_index += 1;
227                continue;
228            }
229
230            let left_top_link = unsafe {
231                // Safety: `link_index` walks 0..buffer.visited.len(), and buffer.visited.len() <= self.links.len().
232                GraphUtil::find_left_top_link(self.links, self.nodes, link_index, &buffer.visited)
233            };
234
235            let link = unsafe {
236                // Safety: `left_top_link` originates from `find_left_top_link`, which only returns
237                // indices in 0..self.links.len(), so this lookup cannot go out of bounds.
238                self.links.get_unchecked(left_top_link)
239            };
240            let is_hole = overlay_rule.is_fill_top(link.fill);
241            let visited_state = [VisitState::HullVisited, VisitState::HoleVisited][is_hole as usize];
242
243            let direction = is_hole == clockwise;
244            let start_data = StartPathData::new(direction, link, left_top_link);
245
246            self.find_contour(
247                &start_data,
248                direction,
249                visited_state,
250                &mut buffer.visited,
251                &mut buffer.points,
252            );
253            let (is_valid, _) = buffer.points.validate(
254                self.options.min_output_area,
255                self.options.preserve_output_collinear,
256            );
257
258            if !is_valid {
259                link_index += 1;
260                continue;
261            }
262
263            output.add_contour(buffer.points.as_slice());
264        }
265    }

296pub(crate) trait GraphContour {
297    fn validate(&mut self, min_output_area: u64, preserve_output_collinear: bool) -> (bool, bool);
298    fn push_node_and_get_other(&mut self, link: &OverlayLink, node_id: usize) -> usize;
299}
300
301impl GraphContour for IntContour {
302    #[inline]
303    fn validate(&mut self, min_output_area: u64, preserve_output_collinear: bool) -> (bool, bool) {
304        let is_modified = if !preserve_output_collinear {
305            self.simplify_contour()
306        } else {
307            false
308        };
309
310        if self.len() < 3 {
311            return (false, is_modified);
312        }
313
314        if min_output_area == 0 {
315            return (true, is_modified);
316        }
317        let area = self.unsafe_area();
318        let abs_area = area.unsigned_abs() >> 1;
319        let is_valid = abs_area >= min_output_area;
320
321        (is_valid, is_modified)
322    }
323
324    #[inline]
325    fn push_node_and_get_other(&mut self, link: &OverlayLink, node_id: usize) -> usize {
326        if link.a.id == node_id {
327            self.push(link.a.point);
328            link.b.id
329        } else {
330            self.push(link.b.point);
331            link.a.id
332        }
333    }
334}

346pub(crate) trait Visit {
347    fn is_visited(&self, index: usize) -> bool;
348    fn is_not_visited(&self, index: usize) -> bool;
349    fn visit_edge(&mut self, index: usize, state: VisitState);
350}
351
352// Safety: every call site creates `visited` slices with one entry per link/node,
353// and they only pass indices directly obtained from those slices. That keeps
354// `index < self.len()` true for the lifetime of the traversal.
355impl Visit for [VisitState] {
356    #[inline(always)]
357    fn is_visited(&self, index: usize) -> bool {
358        unsafe {
359            // SAFETY: callers only pass indices derived from the visited slice itself, so index < len.
360            *self.get_unchecked(index) != VisitState::Unvisited
361        }
362    }
363
364    #[inline(always)]
365    fn is_not_visited(&self, index: usize) -> bool {
366        unsafe {
367            // SAFETY: callers only pass indices derived from the visited slice itself, so index < len.
368            *self.get_unchecked(index) == VisitState::Unvisited
369        }
370    }
371    #[inline(always)]
372    fn visit_edge(&mut self, index: usize, state: VisitState) {
373        unsafe {
374            // SAFETY: callers only pass indices derived from the visited slice itself, so index < len.
375            *self.get_unchecked_mut(index) = state;
376        }
377    }
378}

382impl GraphUtil {
383    /// # Safety
384    /// * `link_index < links.len()`
385    /// * `links[top.a.id]` must exist for every link
386    /// * For bridge nodes, both `bridge[k] < links.len()`
387    /// * `visited` is at least `links.len()` long (or whatever invariant applies)
388    #[inline]
389    pub(crate) unsafe fn find_left_top_link(
390        links: &[OverlayLink],
391        nodes: &[OverlayNode],
392        link_index: usize,
393        visited: &[VisitState],
394    ) -> usize {
395        let top = unsafe {
396            // SAFETY: link_index is always < links.len(); callers either iterate that range or
397            // pull the value from visited, which mirrors links.
398            links.get_unchecked(link_index)
399        };
400        let node = unsafe {
401            // SAFETY: GraphBuilder assigns `a.id` from the index in the `nodes` vector,
402            // so every `id` is guaranteed to lie in 0..nodes.len().
403            nodes.get_unchecked(top.a.id)
404        };
405
406        debug_assert!(top.is_direct());
407
408        match node {
409            OverlayNode::Bridge(bridge) => Self::find_left_top_link_on_bridge(links, bridge),
410            OverlayNode::Cross(indices) => {
411                Self::find_left_top_link_on_indices(links, top, link_index, indices, visited)
412            }
413        }
414    }
415
416    #[inline(always)]
417    fn find_left_top_link_on_indices(
418        links: &[OverlayLink],
419        link: &OverlayLink,
420        link_index: usize,
421        indices: &[usize],
422        visited: &[VisitState],
423    ) -> usize {
424        let mut top_index = link_index;
425        let mut top = link;
426
427        // find most top link
428
429        for &i in indices.iter() {
430            if i == link_index {
431                continue;
432            }
433            let link = unsafe {
434                // SAFETY: indices holds link ids emitted by GraphBuilder, so each i < links.len().
435                links.get_unchecked(i)
436            };
437            if !link.is_direct() || Triangle::is_clockwise_point(top.a.point, top.b.point, link.b.point) {
438                continue;
439            }
440
441            if visited.is_visited(i) {
442                continue;
443            }
444
445            top_index = i;
446            top = link;
447        }
448
449        top_index
450    }
451
452    #[inline(always)]
453    fn find_left_top_link_on_bridge(links: &[OverlayLink], bridge: &[usize; 2]) -> usize {
454        // SAFETY: every bridge index comes straight from GraphBuilder::build_nodes_and_connect_links,
455        // which only records values in 0..links.len(), so the unchecked lookups stay in-bounds.
456        let (l0, l1) = unsafe { (links.get_unchecked(bridge[0]), links.get_unchecked(bridge[1])) };
457        if Triangle::is_clockwise_point(l0.a.point, l0.b.point, l1.b.point) {
458            bridge[0]
459        } else {
460            bridge[1]
461        }
462    }

464    #[inline(always)]
465    pub(crate) fn next_link(
466        links: &[OverlayLink],
467        nodes: &[OverlayNode],
468        link_id: usize,
469        node_id: usize,
470        clockwise: bool,
471        visited: &[VisitState],
472    ) -> usize {
473        let node = unsafe {
474            // SAFETY: all node ids flowing through traversal originate from GraphBuilder,
475            // hence are within `0..nodes.len()`.
476            nodes.get_unchecked(node_id)
477        };
478        match node {
479            OverlayNode::Bridge(bridge) => {
480                if bridge[0] == link_id {
481                    bridge[1]
482                } else {
483                    bridge[0]
484                }
485            }
486            OverlayNode::Cross(indices) => {
487                GraphUtil::find_nearest_link_to(links, link_id, node_id, clockwise, indices, visited)
488            }
489        }
490    }
491
492    // Assumes: `indices` comes from an OverlayNode::Cross built by GraphBuilder,
493    // so every element is a valid index into `links`, and at least one of them is
494    // still unvisited when we enter. The unchecked accesses rely on that invariant.
495    #[inline]
496    fn find_nearest_link_to(
497        links: &[OverlayLink],
498        target_index: usize,
499        node_id: usize,
500        clockwise: bool,
501        indices: &[usize],
502        visited: &[VisitState],
503    ) -> usize {
504        let mut is_first = true;
505        let mut first_index = 0;
506        let mut second_index = usize::MAX;
507        let mut pos = 0;
508        for (i, &link_index) in indices.iter().enumerate() {
509            if visited.is_not_visited(link_index) {
510                if is_first {
511                    first_index = link_index;
512                    is_first = false;
513                } else {
514                    second_index = link_index;
515                    pos = i;
516                    break;
517                }
518            }
519        }
520
521        if second_index == usize::MAX {
522            return first_index;
523        }
524
525        let target = unsafe {
526            // SAFETY: target_index is either the incoming link_id or an entry from indices, both validated.
527            links.get_unchecked(target_index)
528        };
529        let (c, a) = if target.a.id == node_id {
530            (target.a.point, target.b.point)
531        } else {
532            (target.b.point, target.a.point)
533        };
534
535        // more the one vectors
536        let b = unsafe {
537            // SAFETY: first_index originates from indices, so it is within links.
538            links.get_unchecked(first_index)
539        }
540        .other(node_id)
541        .point;
542        let mut vector_solver = NearestVector::new(c, a, b, first_index, clockwise);
543
544        // add second vector
545        vector_solver.add(
546            unsafe {
547                // SAFETY: second_index comes from indices just like first_index.
548                links.get_unchecked(second_index)
549            }
550            .other(node_id)
551            .point,
552            second_index,
553        );
554
555        // check the rest vectors
556        for &link_index in indices.iter().skip(pos + 1) {
557            if visited.is_not_visited(link_index) {
558                let p = unsafe {
559                    // SAFETY: every link_index here is sourced from indices, so it addresses links.
560                    links.get_unchecked(link_index)
561                }
562                .other(node_id)
563                .point;
564                vector_solver.add(p, link_index);
565            }
566        }
567
568        vector_solver.best_id
569    }
