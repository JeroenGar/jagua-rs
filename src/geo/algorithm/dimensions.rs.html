6/// Geometries can have 0, 1, or two dimensions. Or, in the case of an [`empty`](#is_empty)
7/// geometry, a special `Empty` dimensionality.
8///
9/// # Examples
10///
11/// ```
12/// use geo_types::{Point, Rect, line_string};
13/// use geo::dimensions::{HasDimensions, Dimensions};
14///
15/// let point = Point::new(0.0, 5.0);
16/// let line_string = line_string![(x: 0.0, y: 0.0), (x: 5.0, y: 5.0), (x: 0.0, y: 5.0)];
17/// let rect = Rect::new((0.0, 0.0), (10.0, 10.0));
18/// assert_eq!(Dimensions::ZeroDimensional, point.dimensions());
19/// assert_eq!(Dimensions::OneDimensional, line_string.dimensions());
20/// assert_eq!(Dimensions::TwoDimensional, rect.dimensions());
21///
22/// assert!(point.dimensions() < line_string.dimensions());
23/// assert!(rect.dimensions() > line_string.dimensions());
24/// ```
25#[derive(Debug, Clone, Copy, PartialEq, Eq, Ord, PartialOrd)]
26pub enum Dimensions {
27    /// Some geometries, like a `MultiPoint` or `GeometryCollection` may have no elements - thus no
28    /// dimensions. Note that this is distinct from being `ZeroDimensional`, like a `Point`.
29    Empty,
30    /// Dimension of a point
31    ZeroDimensional,
32    /// Dimension of a line or curve
33    OneDimensional,
34    /// Dimension of a surface
35    TwoDimensional,
36}
37
38/// Operate on the dimensionality of geometries.
39pub trait HasDimensions {
40    /// Some geometries, like a `MultiPoint`, can have zero coordinates - we call these `empty`.
41    ///
42    /// Types like `Point` and `Rect`, which have at least one coordinate by construction, can
43    /// never be considered empty.
44    /// ```
45    /// use geo_types::{Point, coord, LineString};
46    /// use geo::HasDimensions;
47    ///
48    /// let line_string = LineString::new(vec![
49    ///     coord! { x: 0., y: 0. },
50    ///     coord! { x: 10., y: 0. },
51    /// ]);
52    /// assert!(!line_string.is_empty());
53    ///
54    /// let empty_line_string: LineString = LineString::empty();
55    /// assert!(empty_line_string.is_empty());
56    ///
57    /// let point = Point::new(0.0, 0.0);
58    /// assert!(!point.is_empty());
59    /// ```
60    fn is_empty(&self) -> bool;
61
62    /// The dimensions of some geometries are fixed, e.g. a Point always has 0 dimensions. However
63    /// for others, the dimensionality depends on the specific geometry instance - for example
64    /// typical `Rect`s are 2-dimensional, but it's possible to create degenerate `Rect`s which
65    /// have either 1 or 0 dimensions.
66    ///
67    /// ## Examples
68    ///
69    /// ```
70    /// use geo_types::{GeometryCollection, Rect, Point};
71    /// use geo::dimensions::{Dimensions, HasDimensions};
72    ///
73    /// // normal rectangle
74    /// let rect = Rect::new((0.0, 0.0), (10.0, 10.0));
75    /// assert_eq!(Dimensions::TwoDimensional, rect.dimensions());
76    ///
77    /// // "rectangle" with zero height degenerates to a line
78    /// let degenerate_line_rect = Rect::new((0.0, 10.0), (10.0, 10.0));
79    /// assert_eq!(Dimensions::OneDimensional, degenerate_line_rect.dimensions());
80    ///
81    /// // "rectangle" with zero height and zero width degenerates to a point
82    /// let degenerate_point_rect = Rect::new((10.0, 10.0), (10.0, 10.0));
83    /// assert_eq!(Dimensions::ZeroDimensional, degenerate_point_rect.dimensions());
84    ///
85    /// // collections inherit the greatest dimensionality of their elements
86    /// let geometry_collection = GeometryCollection::new_from(vec![degenerate_line_rect.into(), degenerate_point_rect.into()]);
87    /// assert_eq!(Dimensions::OneDimensional, geometry_collection.dimensions());
88    ///
89    /// let point = Point::new(10.0, 10.0);
90    /// assert_eq!(Dimensions::ZeroDimensional, point.dimensions());
91    ///
92    /// // An `Empty` dimensionality is distinct from, and less than, being 0-dimensional
93    /// let empty_collection = GeometryCollection::<f32>::empty();
94    /// assert_eq!(Dimensions::Empty, empty_collection.dimensions());
95    /// assert!(empty_collection.dimensions() < point.dimensions());
96    /// ```
97    fn dimensions(&self) -> Dimensions;
98
99    /// The dimensions of the `Geometry`'s boundary, as used by OGC-SFA.
100    ///
101    /// ## Examples
102    ///
103    /// ```
104    /// use geo_types::{GeometryCollection, Rect, Point};
105    /// use geo::dimensions::{Dimensions, HasDimensions};
106    ///
107    /// // a point has no boundary
108    /// let point = Point::new(10.0, 10.0);
109    /// assert_eq!(Dimensions::Empty, point.boundary_dimensions());
110    ///
111    /// // a typical rectangle has a *line* (one dimensional) boundary
112    /// let rect = Rect::new((0.0, 0.0), (10.0, 10.0));
113    /// assert_eq!(Dimensions::OneDimensional, rect.boundary_dimensions());
114    ///
115    /// // a "rectangle" with zero height degenerates to a line, whose boundary is two points
116    /// let degenerate_line_rect = Rect::new((0.0, 10.0), (10.0, 10.0));
117    /// assert_eq!(Dimensions::ZeroDimensional, degenerate_line_rect.boundary_dimensions());
118    ///
119    /// // a "rectangle" with zero height and zero width degenerates to a point,
120    /// // and points have no boundary
121    /// let degenerate_point_rect = Rect::new((10.0, 10.0), (10.0, 10.0));
122    /// assert_eq!(Dimensions::Empty, degenerate_point_rect.boundary_dimensions());
123    ///
124    /// // collections inherit the greatest dimensionality of their elements
125    /// let geometry_collection = GeometryCollection::new_from(vec![degenerate_line_rect.into(), degenerate_point_rect.into()]);
126    /// assert_eq!(Dimensions::ZeroDimensional, geometry_collection.boundary_dimensions());
127    ///
128    /// let geometry_collection = GeometryCollection::<f32>::new_from(vec![]);
129    /// assert_eq!(Dimensions::Empty, geometry_collection.boundary_dimensions());
130    /// ```
131    fn boundary_dimensions(&self) -> Dimensions;
132}

164impl<C: CoordNum> HasDimensions for Line<C> {
165    fn is_empty(&self) -> bool {
166        false
167    }
168
169    fn dimensions(&self) -> Dimensions {
170        if self.start == self.end {
171            // degenerate line is a point
172            Dimensions::ZeroDimensional
173        } else {
174            Dimensions::OneDimensional
175        }
176    }
177
178    fn boundary_dimensions(&self) -> Dimensions {
179        if self.start == self.end {
180            // degenerate line is a point, which has no boundary
181            Dimensions::Empty
182        } else {
183            Dimensions::ZeroDimensional
184        }
185    }

188impl<C: CoordNum> HasDimensions for LineString<C> {
189    fn is_empty(&self) -> bool {
190        self.0.is_empty()
191    }
192
193    fn dimensions(&self) -> Dimensions {
194        if self.0.is_empty() {
195            return Dimensions::Empty;
196        }
197
198        let first = self.0[0];
199        if self.0.iter().any(|&coord| first != coord) {
200            Dimensions::OneDimensional
201        } else {
202            // all coords are the same - i.e. a point
203            Dimensions::ZeroDimensional
204        }
205    }
206
207    /// ```
208    /// use geo_types::line_string;
209    /// use geo::dimensions::{HasDimensions, Dimensions};
210    ///
211    /// let ls = line_string![(x: 0.,  y: 0.), (x: 0., y: 1.), (x: 1., y: 1.)];
212    /// assert_eq!(Dimensions::ZeroDimensional, ls.boundary_dimensions());
213    ///
214    /// let ls = line_string![(x: 0.,  y: 0.), (x: 0., y: 1.), (x: 1., y: 1.), (x: 0., y: 0.)];
215    /// assert_eq!(Dimensions::Empty, ls.boundary_dimensions());
216    ///```
217    fn boundary_dimensions(&self) -> Dimensions {
218        if self.is_closed() {
219            return Dimensions::Empty;
220        }
221
222        match self.dimensions() {
223            Dimensions::Empty | Dimensions::ZeroDimensional => Dimensions::Empty,
224            Dimensions::OneDimensional => Dimensions::ZeroDimensional,
225            Dimensions::TwoDimensional => unreachable!("line_string cannot be 2 dimensional"),
226        }
227    }
228}

230impl<C: CoordNum> HasDimensions for Polygon<C> {
231    fn is_empty(&self) -> bool {
232        self.exterior().is_empty()
233    }
234
235    fn dimensions(&self) -> Dimensions {
236        use crate::CoordsIter;
237        let mut coords = self.exterior_coords_iter();
238
239        let Some(first) = coords.next() else {
240            // No coordinates - the polygon is empty
241            return Dimensions::Empty;
242        };
243
244        let Some(second) = coords.find(|next| *next != first) else {
245            // All coordinates in the polygon are the same point
246            return Dimensions::ZeroDimensional;
247        };
248
249        let Some(_third) = coords.find(|next| *next != first && *next != second) else {
250            // There are only two distinct coordinates in the Polygon - it's collapsed to a line
251            return Dimensions::OneDimensional;
252        };
253
254        Dimensions::TwoDimensional
255    }
256
257    fn boundary_dimensions(&self) -> Dimensions {
258        match self.dimensions() {
259            Dimensions::Empty | Dimensions::ZeroDimensional => Dimensions::Empty,
260            Dimensions::OneDimensional => Dimensions::ZeroDimensional,
261            Dimensions::TwoDimensional => Dimensions::OneDimensional,
262        }
263    }
264}

284impl<C: CoordNum> HasDimensions for MultiLineString<C> {
285    fn is_empty(&self) -> bool {
286        self.iter().all(LineString::is_empty)
287    }
288
289    fn dimensions(&self) -> Dimensions {
290        let mut max = Dimensions::Empty;
291        for line in &self.0 {
292            match line.dimensions() {
293                Dimensions::Empty => {}
294                Dimensions::ZeroDimensional => max = Dimensions::ZeroDimensional,
295                Dimensions::OneDimensional => {
296                    // return early since we know multi line string dimensionality cannot exceed
297                    // 1-d
298                    return Dimensions::OneDimensional;
299                }
300                Dimensions::TwoDimensional => unreachable!("MultiLineString cannot be 2d"),
301            }
302        }
303        max
304    }
305
306    fn boundary_dimensions(&self) -> Dimensions {
307        if self.is_closed() {
308            return Dimensions::Empty;
309        }
310
311        match self.dimensions() {
312            Dimensions::Empty | Dimensions::ZeroDimensional => Dimensions::Empty,
313            Dimensions::OneDimensional => Dimensions::ZeroDimensional,
314            Dimensions::TwoDimensional => unreachable!("line_string cannot be 2 dimensional"),
315        }
316    }
317}

319impl<C: CoordNum> HasDimensions for MultiPolygon<C> {
320    fn is_empty(&self) -> bool {
321        self.iter().all(Polygon::is_empty)
322    }
323
324    fn dimensions(&self) -> Dimensions {
325        let mut max = Dimensions::Empty;
326        for geom in self {
327            let dimensions = geom.dimensions();
328            if dimensions == Dimensions::TwoDimensional {
329                // short-circuit since we know none can be larger
330                return Dimensions::TwoDimensional;
331            }
332            max = max.max(dimensions)
333        }
334        max
335    }
336
337    fn boundary_dimensions(&self) -> Dimensions {
338        match self.dimensions() {
339            Dimensions::Empty | Dimensions::ZeroDimensional => Dimensions::Empty,
340            Dimensions::OneDimensional => Dimensions::ZeroDimensional,
341            Dimensions::TwoDimensional => Dimensions::OneDimensional,
342        }
343    }
344}

346impl<C: GeoNum> HasDimensions for GeometryCollection<C> {
347    fn is_empty(&self) -> bool {
348        if self.0.is_empty() {
349            true
350        } else {
351            self.iter().all(Geometry::is_empty)
352        }
353    }
354
355    fn dimensions(&self) -> Dimensions {
356        let mut max = Dimensions::Empty;
357        for geom in self {
358            let dimensions = geom.dimensions();
359            if dimensions == Dimensions::TwoDimensional {
360                // short-circuit since we know none can be larger
361                return Dimensions::TwoDimensional;
362            }
363            max = max.max(dimensions)
364        }
365        max
366    }
367
368    fn boundary_dimensions(&self) -> Dimensions {
369        let mut max = Dimensions::Empty;
370        for geom in self {
371            let d = geom.boundary_dimensions();
372
373            if d == Dimensions::OneDimensional {
374                return Dimensions::OneDimensional;
375            }
376
377            max = max.max(d);
378        }
379        max
380    }
381}

383impl<C: CoordNum> HasDimensions for Rect<C> {
384    fn is_empty(&self) -> bool {
385        false
386    }
387
388    fn dimensions(&self) -> Dimensions {
389        if self.min() == self.max() {
390            // degenerate rectangle is a point
391            Dimensions::ZeroDimensional
392        } else if self.min().x == self.max().x || self.min().y == self.max().y {
393            // degenerate rectangle is a line
394            Dimensions::OneDimensional
395        } else {
396            Dimensions::TwoDimensional
397        }
398    }
399
400    fn boundary_dimensions(&self) -> Dimensions {
401        match self.dimensions() {
402            Dimensions::Empty => {
403                unreachable!("even a degenerate rect should be at least 0-Dimensional")
404            }
405            Dimensions::ZeroDimensional => Dimensions::Empty,
406            Dimensions::OneDimensional => Dimensions::ZeroDimensional,
407            Dimensions::TwoDimensional => Dimensions::OneDimensional,
408        }
409    }
410}

412impl<C: GeoNum> HasDimensions for Triangle<C> {
413    fn is_empty(&self) -> bool {
414        false
415    }
416
417    fn dimensions(&self) -> Dimensions {
418        use crate::Kernel;
419        if Collinear == C::Ker::orient2d(self.v1(), self.v2(), self.v3()) {
420            if self.v1() == self.v2() && self.v2() == self.v3() {
421                // degenerate triangle is a point
422                Dimensions::ZeroDimensional
423            } else {
424                // degenerate triangle is a line
425                Dimensions::OneDimensional
426            }
427        } else {
428            Dimensions::TwoDimensional
429        }
430    }
431
432    fn boundary_dimensions(&self) -> Dimensions {
433        match self.dimensions() {
434            Dimensions::Empty => {
435                unreachable!("even a degenerate triangle should be at least 0-dimensional")
436            }
437            Dimensions::ZeroDimensional => Dimensions::Empty,
438            Dimensions::OneDimensional => Dimensions::ZeroDimensional,
439            Dimensions::TwoDimensional => Dimensions::OneDimensional,
440        }
441    }
442}
