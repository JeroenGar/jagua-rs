1//! Implements `Intersects` for MonotoneChain Geometry types
2//!
3//! Has optimized implementations when both sides are [`MonotoneChain`] backed
4//! Falls back to base geometry when [`MonotoneChain`] is not supported on one side
5
6// `Geom` intersects `MonotoneChain` and `MonotoneChainSegment` requires adding a lifetime to the macro
7// duplicated from `intersects/mod.rs` but with additional lifetime generic
8macro_rules! symmetric_intersects_impl {
9    ($t:ty, $k:ty) => {
10        impl<'a, T> $crate::Intersects<$k> for $t
11        where
12            $k: $crate::Intersects<$t>,
13            T: GeoNum,
14        {
15            fn intersects(&self, rhs: &$k) -> bool {
16                rhs.intersects(self)
17            }
18        }
19    };
20}
21
22macro_rules! chains_intersects_impl {
23    ($t:ty, $k:ty) => {
24        impl<'a, T> $crate::Intersects<$k> for $t
25        where
26            T: GeoNum,
27        {
28            fn intersects(&self, rhs: &$k) -> bool {
29                use crate::MonotoneChains;
30                rhs.chains()
31                    .any(|c| self.chains().any(|c2| c.intersects(c2)))
32            }
33        }
34    };
35}

53    #[test]
54    fn test_intersects_edgecase() {
55        let pt: Point<f64> = wkt! {POINT(0 0)}.convert();
56
57        let ls0: LineString<f64> = LineString::empty();
58        let ls1: LineString<f64> = wkt! {LINESTRING(0 0)}.convert();
59        let ls2: LineString<f64> = wkt! {LINESTRING(0 0,1 1)}.convert();
60
61        let mcs0: MonotoneChain<f64> = (&ls0).into();
62        let mcs1: MonotoneChain<f64> = (&ls1).into();
63        let mcs2: MonotoneChain<f64> = (&ls2).into();
64
65        assert_eq!(ls0.intersects(&pt.0), mcs0.intersects(&pt.0));
66        assert_eq!(ls1.intersects(&pt.0), ls1.relate(&pt).is_intersects());
67        assert_eq!(ls1.intersects(&pt.0), mcs1.intersects(&pt.0));
68        assert_eq!(ls2.intersects(&pt.0), mcs2.intersects(&pt.0));
69    }
