5/// The [Graham's scan] algorithm to compute the convex hull
6/// of a collection of points. This algorithm is less
7/// performant than the quick hull, but allows computing all
8/// the unique points on the convex hull, as opposed to a strict
9/// convex hull that does not include collinear points.
10///
11/// # References
12///
13/// Graham, R.L. (1972). ["An Efficient Algorithm for
14/// Determining the Convex Hull of a Finite Planar
15/// Set"](http://www.math.ucsd.edu/~ronspubs/72_10_convex_hull.pdf)
16/// (PDF). \
17/// Information Processing Letters. 1 (4): 132–133.
18/// [doi:10.1016/0020-0190(72)90045-2](https://doi.org/10.1016%2F0020-0190%2872%2990045-2)
19///
20/// [Graham's scan]: //en.wikipedia.org/wiki/Graham_scan
21pub fn graham_hull<T>(mut points: &mut [Coord<T>], include_on_hull: bool) -> LineString<T>
22where
23    T: GeoNum,
24{
25    if points.len() < 4 {
26        // Nothing to build with fewer than four points.
27        return trivial_hull(points, include_on_hull);
28    }
29
30    // Allocate output vector
31    let mut output = Vec::with_capacity(points.len());
32
33    // Find lexicographically least point and add to hull
34    use crate::utils::least_index;
35    use std::cmp::Ordering;
36    let min_idx = least_index(points);
37    let head = swap_with_first_and_remove(&mut points, min_idx);
38    output.push(*head);
39
40    // Sort rest of the points by angle it makes with head
41    // point. If two points are collinear with head, we sort
42    // by distance. We use kernel predicates here.
43    let cmp = |q: &Coord<T>, r: &Coord<T>| match T::Ker::orient2d(*q, *head, *r) {
44        Orientation::CounterClockwise => Ordering::Greater,
45        Orientation::Clockwise => Ordering::Less,
46        Orientation::Collinear => {
47            let dist1 = T::Ker::square_euclidean_distance(*head, *q);
48            let dist2 = T::Ker::square_euclidean_distance(*head, *r);
49            dist1.partial_cmp(&dist2).unwrap()
50        }
51    };
52    points.sort_unstable_by(cmp);
53
54    for pt in points.iter() {
55        while output.len() > 1 {
56            let len = output.len();
57            if output[len - 1] == *pt {
58                output.pop();
59                continue;
60            }
61            match T::Ker::orient2d(output[len - 2], output[len - 1], *pt) {
62                Orientation::CounterClockwise => {
63                    break;
64                }
65                Orientation::Clockwise => {
66                    output.pop();
67                }
68                Orientation::Collinear => {
69                    if include_on_hull {
70                        break;
71                    } else {
72                        output.pop();
73                    }
74                }
75            }
76        }
77        // Corner case: if the lex. least point added before
78        // this loop is repeated, then we should not end up
79        // adding it here (because output.len() == 1 in the
80        // first iteration)
81        if include_on_hull || pt != output.last().unwrap() {
82            output.push(*pt);
83        }
84    }
85
86    // Close and output the line string
87    let mut output = LineString::new(output);
88    output.close();
89    output
90}

101    fn brute_force_convexity<T: GeoNum>(initial: &[Coord<T>]) -> bool {
102        // for every adjacent pair of points, all other points must be on the same side of the line formed by the two points
103        // the side returned for all must be the same
104
105        let mut k: Vec<Orientation> = initial
106            .windows(2)
107            .flat_map(|w| initial.iter().map(|pt| T::Ker::orient2d(w[0], w[1], *pt)))
108            .filter(|o| o != &Orientation::Collinear)
109            .collect();
110        k.dedup();
111
112        k.len() == 1
113    }

115    fn test_convexity<T: GeoNum>(mut initial: Vec<Coord<T>>) {
116        let hull = graham_hull(&mut initial, false);
117        assert!(brute_force_convexity(&hull.0));
118        assert!(hull.is_strictly_ccw_convex());
119        let hull = graham_hull(&mut initial, true);
120        assert!(brute_force_convexity(&hull.0));
121        assert!(hull.is_ccw_convex());
122    }
