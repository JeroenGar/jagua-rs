1use crate::GeoFloat;
2use geo_types::Line;
3use std::rc::Rc;
4use std::sync::Arc;

6/// A 1-dimensional finite line used as input to [`Intersections`](super::Intersections).
7///
8/// This is implemented by [`Line`], but you can implement it on your own
9/// type if you'd like to associate some other data with it.
10///
11/// # Cloning
12///
13/// Note that for usage with the `Intersections` iterator, the type must
14/// also impl. `Clone`. If the custom type is not cheap to clone, use
15/// either a reference to the type, a [`Rc`] or an [`Arc`]. All these
16/// are supported via blanket trait implementations.
17pub trait Cross {
18    /// Scalar used by the line coordinates.
19    type Scalar: GeoFloat;
20
21    /// The geometry associated with this type.
22    fn line(&self) -> Line<Self::Scalar>;
23}
24
25impl<T: GeoFloat> Cross for Line<T> {
26    type Scalar = T;
27
28    fn line(&self) -> Line<Self::Scalar> {
29        *self
30    }
31}
32
33impl<C: Cross> Cross for &C {
34    type Scalar = C::Scalar;
35
36    fn line(&self) -> Line<Self::Scalar> {
37        C::line(*self)
38    }
39}
40
41macro_rules! blanket_impl_smart_pointer {
42    ($ty:ty) => {
43        impl<T: Cross> Cross for $ty {
44            type Scalar = T::Scalar;
45
46            fn line(&self) -> Line<Self::Scalar> {
47                T::line(self)
48            }
49        }
50    };
51}
52blanket_impl_smart_pointer!(Box<T>);
53blanket_impl_smart_pointer!(Rc<T>);
54blanket_impl_smart_pointer!(Arc<T>);
