5/// A [`MonotoneChain`] is a primitive which represents a collection of segments,
6/// each of which is a monotonic sequence of [`Coord<T>`](crate::Coord)
7///
8/// This primitive is not be constructed directly, it is created when building a geometry backed by [`MonotoneChain`]
9#[derive(Debug, Clone)]
10pub struct MonotoneChain<'a, T: CoordNum> {
11    bounding_rect: Option<Rect<T>>,
12    segments: Vec<MonotoneChainSegment<'a, T>>,
13}
14
15impl<'a, T: GeoNum> MonotoneChain<'a, T> {
16    /// Iterate over the segments of the chain
17    pub fn segment_iter(&self) -> impl Iterator<Item = &MonotoneChainSegment<'a, T>> {
18        self.segments.iter()
19    }
20}
21
22impl<'a, T: GeoNum> From<&'a LineString<T>> for MonotoneChain<'a, T> {
23    fn from(linestring: &'a LineString<T>) -> Self {
24        // each segment is a series of coordinates which are montonically increasing/decreasing
25        Self {
26            bounding_rect: linestring.bounding_rect(),
27            segments: MonotoneChainSegmentFactory::new(&linestring.0).collect(),
28        }
29    }
30}
31
32// This is here because instead of with the trait because `bounding_rect` is a private field
33impl<'a, T: GeoNum> BoundingRect<T> for MonotoneChain<'a, T> {
34    type Output = Option<Rect<T>>;
35
36    fn bounding_rect(&self) -> Self::Output {
37        self.bounding_rect
38    }
39}

51        let chain: MonotoneChain<f64> = (&linestring).into();
52        assert_eq!(chain.bounding_rect(), None);

55    #[test]
56    fn test_construct_edge_case() {
57        let linestring: LineString<f64> = wkt! { LINESTRING(0 0) }.convert();
58        let chain: MonotoneChain<f64> = (&linestring).into();
59        assert!(chain.segments.len() == 1);
60
61        let linestring: LineString<f64> = wkt! { LINESTRING(0 0,1 1) }.convert();
62        let chain: MonotoneChain<f64> = (&linestring).into();
63        assert!(chain.segments.len() == 1);
64    }
65
66    #[test]
67    fn test_vertical_horizontal() {
68        let h = LineString::from_iter((0..10).map(|x| coord! { x: x as f64, y: 0. }));
69        let h_r = LineString::from_iter((0..10).rev().map(|x| coord! { x: x as f64, y: 0. }));
70        let v = LineString::from_iter((0..10).map(|y| coord! { x: 0., y: y as f64 }));
71        let v_r = LineString::from_iter((0..10).rev().map(|y| coord! { x: 0., y: y as f64 }));
72
73        let h_chain: MonotoneChain<f64> = (&h).into();
74        let h_rev_chain: MonotoneChain<f64> = (&h_r).into();
75        let v_chain: MonotoneChain<f64> = (&v).into();
76        let v_rev_chain: MonotoneChain<f64> = (&v_r).into();
77
78        assert!(h_chain.segments.len() == 1);
79        assert!(h_rev_chain.segments.len() == 1);
80        assert!(v_chain.segments.len() == 1);
81        assert!(v_rev_chain.segments.len() == 1);
82    }
83
84    #[test]
