12/// An enum representing any possible geometry type backed by [`MonotoneChain`](`crate::MonotoneChain`).
13///
14///  # Example
15///
16/// ```
17/// use std::convert::TryFrom;
18/// use geo::{LineString, line_string, MonotoneChainGeometry, Geometry};
19///
20/// let ls: LineString<f64> = line_string![(x: 0., y: 0.), (x: 10., y: 10.)];
21/// let pe: Geometry = ls.into();
22/// let lg: MonotoneChainGeometry<f64> = (&pe).try_into().expect("failed to convert");
23/// ```
24pub enum MonotoneChainGeometry<'a, T: GeoNum> {
25    LineString(MonotoneChainLineString<'a, T>),
26    MultiLineString(MonotoneChainMultiLineString<'a, T>),
27    Polygon(MonotoneChainPolygon<'a, T>),
28    MultiPolygon(MonotoneChainMultiPolygon<'a, T>),
29}
30
31impl<'a, T: GeoNum> TryFrom<&'a Geometry<T>> for MonotoneChainGeometry<'a, T> {
32    type Error = Error;
33
34    fn try_from(geometry: &'a Geometry<T>) -> Result<Self, Self::Error> {
35        match geometry {
36            Geometry::LineString(g) => Ok(Self::LineString((g).into())),
37            Geometry::MultiLineString(g) => Ok(Self::MultiLineString((g).into())),
38            Geometry::Polygon(g) => Ok(Self::Polygon((g).into())),
39            Geometry::MultiPolygon(g) => Ok(Self::MultiPolygon((g).into())),
40            other => Err(Error::MismatchedGeometry {
41                expected: "LineString, MultiLineString, Polygon, MultiPolygon",
42                found: inner_type_name(other),
43            }),
44        }
45    }
46}
47
48/// Duplicated from geo-types
49fn inner_type_name<T>(geometry: &Geometry<T>) -> &'static str
50where
51    T: CoordNum,
52{
53    match *geometry {
54        Geometry::Point(_) => type_name::<Point<T>>(),
55        Geometry::Line(_) => type_name::<Line<T>>(),
56        Geometry::LineString(_) => type_name::<LineString<T>>(),
57        Geometry::Polygon(_) => type_name::<Polygon<T>>(),
58        Geometry::MultiPoint(_) => type_name::<MultiPoint<T>>(),
59        Geometry::MultiLineString(_) => type_name::<MultiLineString<T>>(),
60        Geometry::MultiPolygon(_) => type_name::<MultiPolygon<T>>(),
61        Geometry::GeometryCollection(_) => type_name::<GeometryCollection<T>>(),
62        Geometry::Rect(_) => type_name::<Rect<T>>(),
63        Geometry::Triangle(_) => type_name::<Triangle<T>>(),
64    }
65}
