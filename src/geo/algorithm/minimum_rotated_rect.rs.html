1use geo_types::{Coord, LineString};
2use num_traits::Bounded;
3
4use crate::{ConvexHull, CoordFloat, GeoFloat, GeoNum, Polygon, algorithm::CoordsIter};
5/// Return the minimum bounding rectangle(MBR) of geometry
6/// reference: <https://en.wikipedia.org/wiki/Minimum_bounding_box>
7/// minimum rotated rect is the rectangle that can enclose all points given
8/// and have smallest area of all enclosing rectangles
9/// the rect can be any-oriented, not only axis-aligned.
10///
11/// # Examples
12///
13/// ```
14/// use geo_types::{line_string, polygon, LineString, Polygon};
15/// use geo::MinimumRotatedRect;
16/// let poly: Polygon<f64> = polygon![(x: 3.3, y: 30.4), (x: 1.7, y: 24.6), (x: 13.4, y: 25.1), (x: 14.4, y: 31.0), (x:3.3, y:30.4)];
17/// let mbr = MinimumRotatedRect::minimum_rotated_rect(&poly).unwrap();
18/// assert_eq!(
19///     mbr.exterior(),
20///     &LineString::from(vec![
21///         (1.6999999999999975, 24.6),
22///         (1.450145836371588, 30.44658742890477),
23///         (14.4, 31.0),
24///         (14.64985416362841, 25.15341257109523),
25///         (1.6999999999999975, 24.6),
26///     ])
27/// );
28/// ```
29pub trait MinimumRotatedRect<T> {
30    type Scalar: GeoNum;
31    fn minimum_rotated_rect(&self) -> Option<Polygon<Self::Scalar>>;
32}
33
34impl<T, G> MinimumRotatedRect<T> for G
35where
36    T: CoordFloat + GeoFloat + GeoNum,
37    G: CoordsIter<Scalar = T>,
38{
39    type Scalar = T;
40
41    fn minimum_rotated_rect(&self) -> Option<Polygon<Self::Scalar>> {
42        let hull = ConvexHull::convex_hull(self);
43
44        // We take unit vectors along and perpendicular to each edge, then use
45        // them to build a rotation matrix and apply it to the convex hull,
46        // keeping track of the best AABB found.
47        //
48        // See also the discussion at
49        // https://gis.stackexchange.com/questions/22895/finding-minimum-area-rectangle-for-given-points/22934
50        let mut min_area = <T as Bounded>::max_value();
51        let mut best_edge = None;
52        let (mut best_min_x, mut best_max_x) = (T::zero(), T::zero());
53        let (mut best_min_y, mut best_max_y) = (T::zero(), T::zero());
54
55        // Pick a hull vertex and translate it to the origin to improve precision
56        let ref_p = hull.exterior().points().next()?;
57
58        for edge in hull.exterior().lines() {
59            let (dx, dy) = edge.delta().x_y();
60            let norm = dx.hypot(dy);
61            if norm.is_zero() {
62                continue;
63            }
64            let edge = (dx / norm, dy / norm);
65
66            let (mut min_x, mut max_x) = (T::zero(), T::zero());
67            let (mut min_y, mut max_y) = (T::zero(), T::zero());
68
69            for point in hull
70                .exterior()
71                .points()
72                .skip(1)
73                .take(hull.exterior().0.len() - 2)
74            {
75                let tx = point.x() - ref_p.x();
76                let ty = point.y() - ref_p.y();
77
78                let x = tx * edge.0 + ty * edge.1;
79                let y = -tx * edge.1 + ty * edge.0;
80
81                min_x = min_x.min(x);
82                max_x = max_x.max(x);
83                min_y = min_y.min(y);
84                max_y = max_y.max(y);
85            }
86
87            let area = (max_x - min_x) * (max_y - min_y);
88            if area < min_area {
89                min_area = area;
90                best_edge = Some(edge);
91                best_min_x = min_x;
92                best_max_x = max_x;
93                best_min_y = min_y;
94                best_max_y = max_y;
95            }
96        }
97
98        if let Some((dx, dy)) = best_edge {
99            let to_world = |x: T, y: T| Coord {
100                x: x * dx + y * -dy + ref_p.x(),
101                y: x * dy + y * dx + ref_p.y(),
102            };
103
104            let p1 = to_world(best_min_x, best_min_y);
105            let p2 = to_world(best_min_x, best_max_y);
106            let p3 = to_world(best_max_x, best_max_y);
107            let p4 = to_world(best_max_x, best_min_y);
108            let rectangle = Polygon::new(LineString(vec![p1, p2, p3, p4, p1]), Vec::new());
109            Some(rectangle)
110        } else {
111            None
112        }
113    }
114}

116#[cfg(test)]
117mod test {
118    use geo_types::{LineString, Polygon, line_string, polygon};
119
120    use crate::MinimumRotatedRect;
121
122    #[test]
123    fn returns_polygon_mbr() {
124        let poly: Polygon<f64> = polygon![(x: 3.3, y: 30.4), (x: 1.7, y: 24.6), (x: 13.4, y: 25.1), (x: 14.4, y: 31.0),(x:3.3,y:30.4)];
125        let mbr = MinimumRotatedRect::minimum_rotated_rect(&poly).unwrap();
126        assert_eq!(
127            mbr.exterior(),
128            &LineString::from(vec![
129                (1.6999999999999975, 24.6),
130                (1.450145836371588, 30.44658742890477),
131                (14.4, 31.0),
132                (14.64985416362841, 25.15341257109523),
133                (1.6999999999999975, 24.6),
134            ])
135        );
136    }
137    #[test]
138    fn returns_linestring_mbr() {
139        let poly: LineString<f64> = line_string![(x: 3.3, y: 30.4), (x: 1.7, y: 24.6), (x: 13.4, y: 25.1), (x: 14.4, y: 31.0)];
140        let mbr = MinimumRotatedRect::minimum_rotated_rect(&poly).unwrap();
141        assert_eq!(
142            mbr.exterior(),
143            &LineString::from(vec![
144                (1.6999999999999975, 24.6),
145                (1.450145836371588, 30.44658742890477),
146                (14.4, 31.0),
147                (14.64985416362841, 25.15341257109523),
148                (1.6999999999999975, 24.6),
149            ])
150        );
151    }
