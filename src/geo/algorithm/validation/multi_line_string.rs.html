7/// A [`MultiLineString`] is valid if each [`LineString`](crate::LineString) in it is valid.
8#[derive(Debug, Clone, PartialEq)]
9pub enum InvalidMultiLineString {
10    /// Which element is invalid, and what was invalid about it.
11    InvalidLineString(GeometryIndex, InvalidLineString),
12}
13
14impl fmt::Display for InvalidMultiLineString {
15    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
16        match self {
17            InvalidMultiLineString::InvalidLineString(idx, err) => {
18                write!(f, "line string at index {} is invalid: {}", idx.0, err)
19            }
20        }
21    }
22}
23
24impl std::error::Error for InvalidMultiLineString {}
25
26impl<F: GeoFloat> Validation for MultiLineString<F> {
27    type Error = InvalidMultiLineString;
28
29    fn visit_validation<T>(
30        &self,
31        mut handle_validation_error: Box<dyn FnMut(Self::Error) -> Result<(), T> + '_>,
32    ) -> Result<(), T> {
33        for (i, line_string) in self.0.iter().enumerate() {
34            line_string.visit_validation(Box::new(&mut |line_string_err| {
35                let err =
36                    InvalidMultiLineString::InvalidLineString(GeometryIndex(i), line_string_err);
37                handle_validation_error(err)
38            }))?;
39        }
40        Ok(())
41    }
42}
