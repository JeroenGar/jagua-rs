16static RAYS_PER_ANGLE: usize = if cfg!(debug_assertions) { 10 } else { 200 };
17static N_ANGLES: usize = if cfg!(debug_assertions) { 4 } else { 90 };
18static N_POINTS_PER_DIMENSION: usize = if cfg!(debug_assertions) { 10 } else { 100 };
19static CLIPPING_TRIM: f32 = 0.999;
20static ACTION_RADIUS_RATIO: f32 = 0.10;

22/// Generates a set of `n` *piers* - line segments fully contained within `shape`.
23/// This function generates them in such a way as to *cover* areas of the `shape` that are
24/// poorly represented by `poles` as well as possible.
25pub fn generate_piers(shape: &SPolygon, n: usize, poles: &[Circle]) -> Result<Vec<Edge>> {
26    if n == 0 {
27        return Ok(vec![]);
28    }
29
30    //Start by creating a set of N_TESTS_PER_ANGLE vertical lines across the bounding box
31    let bbox = shape.bbox;
32    let expanded_bbox = bbox.clone().inflate_to_square();
33    let centroid = shape.centroid();
34    //vertical ray through the origin, positioned relative to the centroid by the transformations
35    let base_ray = Edge::try_new(
36        Point(0.0, -2.0 * expanded_bbox.height()),
37        Point(0.0, 2.0 * expanded_bbox.height()),
38    )
39    .unwrap();
40
41    let transformations =
42        generate_ray_transformations(expanded_bbox, centroid, RAYS_PER_ANGLE, N_ANGLES);
43
44    //transform the base edge by each transformation
45    let rays = transformations
46        .into_iter()
47        .map(|t| base_ray.transform_clone(&t))
48        .collect_vec();
49
50    //clip the lines to the shape
51    let clipped_rays = rays.iter().flat_map(|l| clip(shape, l)).collect_vec();
52    let grid_of_unrepresented_points =
53        generate_unrepresented_point_grid(expanded_bbox, shape, poles, N_POINTS_PER_DIMENSION);
54
55    let mut selected_piers = Vec::new();
56
57    let radius_of_ray_influence = ACTION_RADIUS_RATIO * expanded_bbox.width();
58    let forfeit_distance = f32::sqrt(bbox.width().powi(2) * bbox.height().powi(2));
59
60    for _ in 0..n {
61        let min_distance_selected_rays = min_distances_to_rays(
62            &grid_of_unrepresented_points,
63            &selected_piers,
64            forfeit_distance,
65        );
66        let min_distance_poles =
67            min_distances_to_poles(&grid_of_unrepresented_points, poles, forfeit_distance);
68
69        let loss_values = clipped_rays
70            .iter()
71            .map(|new_ray| {
72                loss_function(
73                    new_ray,
74                    &grid_of_unrepresented_points,
75                    &min_distance_selected_rays,
76                    &min_distance_poles,
77                    radius_of_ray_influence,
78                )
79            })
80            .map(|x| NotNan::new(x).unwrap())
81            .collect_vec();
82
83        let min_loss_ray = clipped_rays
84            .iter()
85            .enumerate()
86            .min_by_key(|(i, _)| loss_values[*i])
87            .map(|(_i, ray)| ray);
88
89        match min_loss_ray {
90            None => bail!("no ray found"),
91            Some(ray) => selected_piers.push(*ray),
92        }
93    }
94    Ok(selected_piers)
95}

97#[allow(clippy::cast_precision_loss)]
98fn generate_ray_transformations(
99    bbox: Rect,
100    centroid: Point,
101    rays_per_angle: usize,
102    n_angles: usize,
103) -> Vec<Transformation> {
104    //translations, relative to the centroid
105    let dx = bbox.width() / rays_per_angle as f32;
106    let translations = (0..rays_per_angle)
107        .map(|i| bbox.x_min - centroid.0 + dx * i as f32)
108        .map(|x| Transformation::from_translation((x, 0.0)))
109        .collect_vec();
110
111    let angles = Array::linspace(0.0, f32::PI(), n_angles + 1).to_vec();
112    let angles_slice = &angles[0..n_angles]; //skip the last angle, which is the same as the first
113
114    //rotate the translations by each angle around the centroid
115    angles_slice
116        .iter()
117        .flat_map(|angle| {
118            translations.iter().cloned().map(move |translation| {
119                translation
120                    .rotate(*angle)
121                    .translate((centroid.0, centroid.1))
122            })
123        })
124        .collect_vec()
125}

127//clips a ray against the border of a polygon, potentially creating multiple "clips"
128fn clip(shape: &SPolygon, ray: &Edge) -> Vec<Edge> {
129    //both ends of the ray should be outside the shape
130    assert!(!shape.collides_with(&ray.start) && !shape.collides_with(&ray.end));
131
132    //collect all intersections of the ray with the shape, sorted by distance to the ray's start
133    let intersections = shape
134        .edge_iter()
135        .filter_map(|edge| edge.collides_at(ray))
136        .sorted_by_key(|p| NotNan::new(ray.start.distance_to(p)).unwrap())
137        .collect_vec();
138
139    //every pair of (sorted) intersections defines a clipped line
140    intersections
141        .chunks(2)
142        .filter_map(|pair| {
143            if pair.len() == 1 {
144                return None;
145            }
146            let start = pair[0];
147            let end = pair[1];
148            if start == end {
149                None
150            } else {
151                Some(Edge::try_new(start, end).unwrap().scale(CLIPPING_TRIM))
152            }
153        })
154        .collect_vec()
155}

157fn generate_unrepresented_point_grid(
158    bbox: Rect,
159    shape: &SPolygon,
160    poles: &[Circle],
161    n_points_per_dimension: usize,
162) -> Vec<Point> {
163    let x_range = Array::linspace(bbox.x_min, bbox.x_max, n_points_per_dimension);
164    let y_range = Array::linspace(bbox.y_min, bbox.y_max, n_points_per_dimension);
165
166    x_range
167        .iter()
168        .flat_map(|x| {
169            y_range
170                .iter()
171                .map(move |y| Point::from((*x, *y))) //create the points
172                .filter(|p| shape.collides_with(p)) //make sure they are in the shape
173                .filter(|p| poles.iter().all(|c| !c.collides_with(p)))
174        })
175        .collect_vec()
176}

178fn loss_function(
179    new_ray: &Edge,
180    point_grid: &[Point],
181    min_distance_to_rays: &[f32],
182    min_distance_to_poles: &[f32],
183    radius_of_ray_influence: f32,
184) -> f32 {
185    //every point in the grid gets a certain score, sum of all these scores is the loss function
186    //the score depends on how close it is to being "represented" by either a pole or a ray
187    //rays have a certain radius of influence, outside which they don't count. Poles have no such radius
188    //the score is the squared distance to the closest ray or pole
189
190    izip!(
191        point_grid.iter(),
192        min_distance_to_rays.iter(),
193        min_distance_to_poles.iter()
194    )
195    .map(|(p, min_distance_to_existing_ray, min_distance_to_pole)| {
196        let distance_to_new_ray = new_ray.distance_to(p);
197
198        let min_distance_to_ray = f32::min(*min_distance_to_existing_ray, distance_to_new_ray);
199
200        if min_distance_to_ray < radius_of_ray_influence {
201            f32::min(*min_distance_to_pole, min_distance_to_ray)
202        } else {
203            *min_distance_to_pole
204        }
205    })
206    .map(|d| d.powi(2))
207    .sum()
208}

210fn min_distances_to_rays(points: &[Point], rays: &[Edge], forfeit_distance: f32) -> Vec<f32> {
211    points
212        .iter()
213        .map(|p| {
214            rays.iter()
215                .map(|r| r.distance_to(p))
216                .fold(forfeit_distance, f32::min)
217        })
218        .collect_vec()
219}
220
221fn min_distances_to_poles(points: &[Point], poles: &[Circle], forfeit_distance: f32) -> Vec<f32> {
222    points
223        .iter()
224        .map(|p| {
225            poles
226                .iter()
227                .map(|c| c.distance_to(p))
228                .fold(forfeit_distance, f32::min)
229        })
230        .collect_vec()
231}

237    #[test]
238    fn piers_are_translation_invariant() {
239        let l_shape = |(dx, dy): (f32, f32)| {
240            let points = [
241                (-2.25, -2.5),
242                (3.75, -2.5),
243                (3.75, -0.5),
244                (0.75, -0.5),
245                (0.75, 3.5),
246                (-2.25, 3.5),
247            ];
248            SPolygon::new(points.map(|(x, y)| Point(x + dx, y + dy)).to_vec()).unwrap()
249        };
250        let offset = (100.0, 50.0);
251        let centered = generate_piers(&l_shape((0.0, 0.0)), 2, &[]).unwrap();
252        let shifted = generate_piers(&l_shape(offset), 2, &[]).unwrap();
253
254        let t = Transformation::from_translation(offset);
255        for (c, s) in centered.iter().zip(&shifted) {
256            let expected = c.transform_clone(&t);
257            assert!(
258                expected.start.distance_to(&s.start) < 1e-3,
259                "{expected:?} != {s:?}"
260            );
261            assert!(
262                expected.end.distance_to(&s.end) < 1e-3,
263                "{expected:?} != {s:?}"
264            );
265        }
266    }
