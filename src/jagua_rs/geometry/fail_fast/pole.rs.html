12/// Generates a set of non-overlapping poles for a shape according to specified coverage limits.
13///
14/// Each new pole is the largest one that fits outside the earlier poles, so the returned order is
15/// largest-remaining-first. See [`compute_pole`] for details on what a pole is.
16/// Generation also stops when no further pole is found, so a thin shape can have fewer poles than the limits allow.
17#[must_use]
18pub fn generate_surrogate_poles(shape: &SPolygon, n_pole_limits: &[(usize, f32)]) -> Vec<Circle> {
19    let mut all_poles = vec![shape.poi];
20    let mut total_pole_area = shape.poi.area();
21
22    //Generate the poles until one of the pole number / coverage limits is reached
23    loop {
24        let current_coverage = total_pole_area / shape.area;
25
26        //check if any limit in the number of poles is reached at this coverage
27        let active_pole_limit = n_pole_limits
28            .iter()
29            .filter(|(_, coverage_threshold)| current_coverage > *coverage_threshold)
30            .min_by_key(|(n_poles, _)| *n_poles)
31            .map(|(n_poles, _)| n_poles);
32
33        if let Some(active_pole_limit) = active_pole_limit
34            && all_poles.len() >= *active_pole_limit
35        {
36            //stop generating if we are above the limit
37            break;
38        }
39
40        //Unlike the pole of inaccessibility, these poles are optional: they are only searched at the default depth,
41        //because searching deeper for each of them is very slow in thin shapes.
42        let Some(next) = search_pole(shape, &all_poles, POI_TREE_DEPTHS[0]) else {
43            //in a thin shape, the search can run out of room for poles before the limits are reached
44            debug!("[POLE] no further pole found after {}", all_poles.len());
45            break;
46        };
47        total_pole_area += next.area();
48        all_poles.push(next);
49        assert!(
50            all_poles.len() < 1000,
51            "More than 1000 poles were generated, please check the SPSurrogateConfig"
52        );
53    }
54    all_poles
55}

57/// Computes the *pole* - the largest circle which is both inside of `shape` while being outside all other `poles`.
58/// Closely related to [Pole of Inaccessibility (PoI)](https://en.wikipedia.org/wiki/Pole_of_inaccessibility),
59/// and inspired by Mapbox's [`polylabel`](https://github.com/mapbox/polylabel) algorithm.
60///
61/// A thin shape can lie between the centers of all nodes of a shallow search tree,
62/// so the search is repeated deeper until a pole is found.
63/// Normal shapes find their pole at the first depth and pay nothing extra.
64pub fn compute_pole(shape: &SPolygon, poles: &[Circle]) -> Result<Circle> {
65    POI_TREE_DEPTHS
66        .iter()
67        .find_map(|&depth| search_pole(shape, poles, depth))
68        .ok_or(anyhow!(
69            "no pole found with up to {} levels of recursion. Please check the input shape: {:?}",
70            POI_TREE_DEPTHS[POI_TREE_DEPTHS.len() - 1],
71            shape.vertices
72        ))
73}

75/// Depths of the successive pole searches in [`compute_pole`].
76/// Each level halves the size of the nodes, the root being the square around the shape's bounding box.
77/// Beyond about 24 levels, `f32` cannot represent the nodes' midpoints anymore.
78const POI_TREE_DEPTHS: [usize; 4] = [10, 14, 18, 22];
79
80fn search_pole(shape: &SPolygon, poles: &[Circle], depth: usize) -> Option<Circle> {
81    let square_bbox = shape.bbox.inflate_to_square();
82    let root = POINode::new(square_bbox, depth, shape, poles);
83    let mut queue = VecDeque::from([root]);
84    let mut best: Option<Circle> = None;
85    let distance = |circle: &Option<Circle>| circle.as_ref().map_or(0.0, |c| c.radius);
86
87    while let Some(node) = queue.pop_front() {
88        //check if better than current best
89        if node.distance > distance(&best) {
90            best = Some(Circle::try_new(node.bbox.centroid(), node.distance).unwrap());
91        }
92
93        //see if worth it to split
94        if node.distance_upperbound() > distance(&best)
95            && let Some(children) = node.split(shape, poles)
96        {
97            queue.extend(children);
98        }
99    }
100    best
101}

103struct POINode {
104    pub level: usize,
105    pub bbox: Rect,
106    pub radius: f32,
107    pub distance: f32,
108}
109
110impl POINode {
111    fn new(bbox: Rect, level: usize, poly: &SPolygon, poles: &[Circle]) -> Self {
112        let radius = bbox.diameter() / 2.0;
113
114        let centroid_inside = poly.collides_with(&bbox.centroid())
115            && poles.iter().all(|c| !c.collides_with(&bbox.centroid()));
116
117        let distance = {
118            let distance_to_edges = poly.edge_iter().map(|e| e.distance_to(&bbox.centroid()));
119
120            let distance_to_poles = poles
121                .iter()
122                .map(|c| c.separation_distance(&bbox.centroid()).1);
123
124            let distance_to_border = distance_to_edges
125                .chain(distance_to_poles)
126                .fold(f32::MAX, f32::min);
127
128            //if the centroid is outside, distance is counted negative
129            if centroid_inside {
130                distance_to_border
131            } else {
132                -distance_to_border
133            }
134        };
135
136        Self {
137            level,
138            bbox,
139            radius,
140            distance,
141        }
142    }
143
144    fn split(&self, poly: &SPolygon, poles: &[Circle]) -> Option<[POINode; 4]> {
145        let Point(m_x, m_y) = self.bbox.centroid();
146        let Rect {
147            x_min,
148            y_min,
149            x_max,
150            y_max,
151        } = self.bbox;
152        //far from the origin, the midpoint can round onto a side, which would leave a quadrant without area
153        let splittable = x_min < m_x && m_x < x_max && y_min < m_y && m_y < y_max;
154        match self.level {
155            0 => None,
156            _ if !splittable => None,
157            _ => Some(
158                self.bbox
159                    .quadrants()
160                    .map(|qd| POINode::new(qd, self.level - 1, poly, poles)),
161            ),
162        }
163    }
164
165    fn distance_upperbound(&self) -> f32 {
166        self.radius + self.distance
167    }
168}
