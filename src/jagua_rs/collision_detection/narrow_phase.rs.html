9/// Bounding boxes of a polygon's edges in flat arrays, so they can be culled 64 at a time.
10/// Box `i` bounds [`SPolygon::edge`]`(i)`. Cloning shares the boxes.
11#[derive(Clone, Debug)]
12pub(crate) struct EdgeBoxes {
13    /// The `x_min`, `y_min`, `x_max` and `y_max` of all boxes, one array after the other
14    bounds: Arc<[f32]>,
15}
16
17impl EdgeBoxes {
18    pub fn new(poly: &SPolygon) -> Self {
19        let bounds = [Edge::x_min, Edge::y_min, Edge::x_max, Edge::y_max]
20            .iter()
21            .flat_map(|bound| poly.edge_iter().map(move |e| bound(&e)))
22            .collect();
23        Self { bounds }
24    }
25
26    /// Calls `f` with every edge of `poly` whose bounding box overlaps `r`, until it returns `true`.
27    /// Returns whether it did.
28    #[inline(always)]
29    pub fn any_edge_overlapping(
30        &self,
31        poly: &SPolygon,
32        r: Rect,
33        mut f: impl FnMut(&Edge) -> bool,
34    ) -> bool {
35        let vertices = poly.vertices();
36        let n = vertices.len();
37        debug_assert_eq!(self.bounds.len(), 4 * n);
38        let (x_min, rest) = self.bounds.split_at(n);
39        let (y_min, rest) = rest.split_at(n);
40        let (x_max, y_max) = rest.split_at(n);
41        for start in (0..n).step_by(64) {
42            let end = (start + 64).min(n);
43            let mut mask = rect_mask(
44                &x_min[start..end],
45                &y_min[start..end],
46                &x_max[start..end],
47                &y_max[start..end],
48                r,
49            );
50            while mask != 0 {
51                let i = start + mask.trailing_zeros() as usize;
52                mask &= mask - 1;
53                let edge = Edge {
54                    start: vertices[i],
55                    end: vertices[if i + 1 == n { 0 } else { i + 1 }],
56                };
57                if f(&edge) {
58                    return true;
59                }
60            }
61        }
62        false
63    }
64}

66/// A polygon's edges grouped in chains that run monotonically in y, each with a bounding box.
67/// A horizontal ray crosses a y-monotone chain at most once, so a point-in-polygon test only
68/// inspects the edges of chains whose box contains the point.
69#[derive(Clone, Debug)]
70pub(crate) struct EdgeChains {
71    /// All edges, chain after chain
72    edges: Vec<Edge>,
73    chains: Vec<Chain>,
74}
75
76#[derive(Clone, Debug)]
77struct Chain {
78    /// The chain's edges in [`EdgeChains::edges`]
79    range: std::ops::Range<usize>,
80    bbox: Rect,
81    /// y of the chain's first and last vertex
82    y_ends: (f32, f32),
83}
84
85/// Longest chain, so diagonal runs keep small boxes
86const MAX_CHAIN_EDGES: usize = 16;
87
88impl EdgeChains {
89    pub fn new(poly: &SPolygon) -> Self {
90        let edges = poly.edge_iter().collect::<Vec<_>>();
91        let mut chains: Vec<Chain> = vec![];
92        // Direction in y of the last chain: 1 up, -1 down, 0 still undecided
93        let mut dir = 0;
94        for (i, e) in edges.iter().enumerate() {
95            let d = match e.end.1.total_cmp(&e.start.1) {
96                std::cmp::Ordering::Greater => 1,
97                std::cmp::Ordering::Less => -1,
98                std::cmp::Ordering::Equal => 0,
99            };
100            match chains.last_mut() {
101                Some(c) if c.range.len() < MAX_CHAIN_EDGES && (d == 0 || dir == 0 || d == dir) => {
102                    c.range.end = i + 1;
103                    c.bbox = Rect::bounding_rect(c.bbox, e.bbox());
104                    c.y_ends.1 = e.end.1;
105                    if dir == 0 {
106                        dir = d;
107                    }
108                }
109                _ => {
110                    chains.push(Chain {
111                        range: i..i + 1,
112                        bbox: e.bbox(),
113                        y_ends: (e.start.1, e.end.1),
114                    });
115                    dir = d;
116                }
117            }
118        }
119        Self { edges, chains }
120    }
121
122    /// Calls `f` with every edge whose bounding box overlaps `r`, until it returns `true`.
123    /// Returns whether it did.
124    #[inline(always)]
125    pub fn any_edge_overlapping(&self, r: Rect, mut f: impl FnMut(&Edge) -> bool) -> bool {
126        for c in &self.chains {
127            if !c.bbox.collides_with(&r) {
128                continue;
129            }
130            for e in &self.edges[c.range.clone()] {
131                if e.bbox().collides_with(&r) && f(e) {
132                    return true;
133                }
134            }
135        }
136        false
137    }
138
139    /// Even-odd point-in-polygon test with a ray towards +x. Chains entirely right of the point
140    /// count from their end points alone; only chains whose box contains the point test edges.
141    pub fn contains(&self, p: Point) -> bool {
142        let Point(px, py) = p;
143        let mut inside = false;
144        for c in &self.chains {
145            if py < c.bbox.y_min || py > c.bbox.y_max || px > c.bbox.x_max {
146                continue;
147            }
148            if px < c.bbox.x_min {
149                // The ray crosses the chain once if its ends straddle the ray
150                if (c.y_ends.0 > py) != (c.y_ends.1 > py) {
151                    inside = !inside;
152                }
153                continue;
154            }
155            for e in &self.edges[c.range.clone()] {
156                let (a, b) = (e.start, e.end);
157                if (a.1 > py) != (b.1 > py) && px < (b.0 - a.0) * (py - a.1) / (b.1 - a.1) + a.0 {
158                    inside = !inside;
159                }
160            }
161        }
162        inside
163    }
164}

166/// A set of edges that can be culled by bounding box.
167pub(crate) trait EdgeSet {
168    /// Calls `f` with every edge whose bounding box overlaps `r`, until it returns `true`.
169    /// Returns whether it did.
170    fn any_edge_overlapping(&self, r: Rect, f: impl FnMut(&Edge) -> bool) -> bool;
171}
172
173impl EdgeSet for EdgeChains {
174    #[inline(always)]
175    fn any_edge_overlapping(&self, r: Rect, f: impl FnMut(&Edge) -> bool) -> bool {
176        EdgeChains::any_edge_overlapping(self, r, f)
177    }
178}

180/// Whether an edge of `poly` intersects an edge of `edges`.
181/// Only edges whose bounding box overlaps `r` are considered on both sides.
182#[inline(always)]
183pub fn edges_intersect(poly: &SPolygon, r: Rect, edges: &impl EdgeSet) -> bool {
184    // Edges of `poly` overlapping `r`, tested against the other side 64 at a time
185    let mut buf = [Edge {
186        start: Point(0.0, 0.0),
187        end: Point(0.0, 0.0),
188    }; 64];
189    let mut n = 0;
190    let vertices = poly.vertices();
191    let mut prev = *vertices.last().unwrap();
192    let mut prev_code = outcode(prev, r);
193    for &cur in vertices {
194        let code = outcode(cur, r);
195        // An edge can only overlap `r` if its end points are not beyond the same side
196        if prev_code & code == 0 {
197            buf[n] = Edge {
198                start: prev,
199                end: cur,
200            };
201            n += 1;
202            if n == buf.len() {
203                if edges.any_edge_overlapping(r, |e| buf.iter().any(|b| b.collides_with(e))) {
204                    return true;
205                }
206                n = 0;
207            }
208        }
209        (prev, prev_code) = (cur, code);
210    }
211    n > 0 && edges.any_edge_overlapping(r, |e| buf[..n].iter().any(|b| b.collides_with(e)))
212}

214/// Whether `poly` contains `p`, decided exactly as [`SPolygon::collides_with`] does, with
215/// `edges` the edges of `poly`. The ray cast only visits edges whose bounding box reaches the
216/// ray, widened in y by the tolerance of the approximate vertex test, since no other edge can
217/// count as a crossing.
218#[inline(always)]
219pub fn poly_contains(poly: &SPolygon, edges: &impl EdgeSet, p: Point) -> bool {
220    if !poly.bbox.collides_with(&p) {
221        return false;
222    }
223    let ray = Edge {
224        start: p,
225        end: Point(poly.bbox.x_max + poly.bbox.width(), p.1),
226    };
227    // At least float_cmp's default margin: f32::EPSILON absolute or 4 ulps relative
228    let margin = f32::EPSILON + 8.0 * f32::EPSILON * p.1.abs();
229    let r = Rect {
230        x_min: p.0,
231        y_min: p.1 - margin,
232        x_max: f32::INFINITY,
233        y_max: p.1 + margin,
234    };
235    let mut crossings = 0;
236    edges.any_edge_overlapping(r, |e| {
237        crossings += usize::from(ray_crosses(e, &p, &ray));
238        false
239    });
240    crossings % 2 == 1
241}

243/// Whether `c` touches an edge of `edges`.
244#[inline(always)]
245pub fn circle_touches_edges(c: &Circle, edges: &impl EdgeSet) -> bool {
246    let sq_radius = c.radius.powi(2);
247    edges.any_edge_overlapping(c.bbox(), |e| e.sq_distance_to(&c.center) <= sq_radius)
248}

250/// Intersection of two rectangles, or `None` if they are disjoint. Touching counts as overlap.
251#[inline(always)]
252pub fn overlap(a: Rect, b: Rect) -> Option<Rect> {
253    let r = Rect {
254        x_min: a.x_min.max(b.x_min),
255        y_min: a.y_min.max(b.y_min),
256        x_max: a.x_max.min(b.x_max),
257        y_max: a.y_max.min(b.y_max),
258    };
259    (r.x_min <= r.x_max && r.y_min <= r.y_max).then_some(r)
260}

262/// Cohen-Sutherland outcode of `p` relative to `r`: one bit per side of `r` that `p` lies beyond.
263#[inline(always)]
264fn outcode(p: Point, r: Rect) -> u8 {
265    u8::from(p.0 < r.x_min)
266        | (u8::from(p.0 > r.x_max) << 1)
267        | (u8::from(p.1 < r.y_min) << 2)
268        | (u8::from(p.1 > r.y_max) << 3)
269}
