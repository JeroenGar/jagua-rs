4use crate::collision_detection::hazards::HazKey;
5use crate::geometry::primitives::Rect;

7/// Bounding boxes of hazards in flat arrays sorted by `x_min`, tested 64 at a time. A query
8/// stops at the first box starting right of it and skips the chunks of 64 boxes that all end left
9/// of it.
10#[derive(Clone, Debug, Default)]
11pub(crate) struct HazardIndex {
12    keys: Vec<HazKey>,
13    x_min: Vec<f32>,
14    y_min: Vec<f32>,
15    x_max: Vec<f32>,
16    y_max: Vec<f32>,
17    /// Largest `x_max` of each chunk of 64 consecutive boxes
18    chunk_x_max: Vec<f32>,
19}
20
21impl HazardIndex {
22    pub fn insert(&mut self, hkey: HazKey, bbox: Rect) {
23        let i = self.x_min.partition_point(|&x| x <= bbox.x_min);
24        self.keys.insert(i, hkey);
25        self.x_min.insert(i, bbox.x_min);
26        self.y_min.insert(i, bbox.y_min);
27        self.x_max.insert(i, bbox.x_max);
28        self.y_max.insert(i, bbox.y_max);
29        self.update_chunks(i / 64);
30    }
31
32    /// Removes the hazard with `hkey`, inserted with `bbox`.
33    pub fn remove(&mut self, hkey: HazKey, bbox: Rect) {
34        let start = self.x_min.partition_point(|&x| x < bbox.x_min);
35        let i = start
36            + self.keys[start..]
37                .iter()
38                .position(|&k| k == hkey)
39                .expect("hazard not in index");
40        self.keys.remove(i);
41        self.x_min.remove(i);
42        self.y_min.remove(i);
43        self.x_max.remove(i);
44        self.y_max.remove(i);
45        self.update_chunks(i / 64);
46    }
47
48    /// Recomputes the bounds of the chunks from `first` on, whose boxes have shifted.
49    fn update_chunks(&mut self, first: usize) {
50        self.chunk_x_max.truncate(first);
51        let maxima = self.x_max[first * 64..]
52            .chunks(64)
53            .map(|chunk| chunk.iter().copied().fold(f32::NEG_INFINITY, f32::max));
54        self.chunk_x_max.extend(maxima);
55    }
56
57    /// Calls `f` for every hazard whose bounding box overlaps `r` (borders included),
58    /// until it returns `true`. Returns whether it did.
59    #[inline(always)]
60    pub fn any_overlapping(&self, r: Rect, mut f: impl FnMut(HazKey) -> bool) -> bool {
61        self.any_chunk(r, |start, mut mask| {
62            while mask != 0 {
63                let i = start + mask.trailing_zeros() as usize;
64                mask &= mask - 1;
65                if f(self.keys[i]) {
66                    return true;
67                }
68            }
69            false
70        })
71    }
72
73    /// Like [`Self::any_overlapping`] for several rectangles, all inside `union`: the boxes are
74    /// scanned once with `union`, then the candidates are tested per rectangle.
75    /// `f` receives the rectangle's position in `rects`.
76    #[inline(always)]
77    pub fn any_overlapping_each(
78        &self,
79        union: Rect,
80        rects: &(impl Iterator<Item = Rect> + Clone),
81        mut f: impl FnMut(usize, HazKey) -> bool,
82    ) -> bool {
83        self.any_chunk(union, |start, candidates| {
84            for (ri, r) in rects.clone().enumerate() {
85                let mut mask = candidates;
86                while mask != 0 {
87                    let i = start + mask.trailing_zeros() as usize;
88                    mask &= mask - 1;
89                    let overlaps = self.x_min[i] <= r.x_max
90                        && self.x_max[i] >= r.x_min
91                        && self.y_min[i] <= r.y_max
92                        && self.y_max[i] >= r.y_min;
93                    if overlaps && f(ri, self.keys[i]) {
94                        return true;
95                    }
96                }
97            }
98            false
99        })
100    }
101
102    /// The hazards in the index with their bounding boxes, sorted by `x_min`.
103    pub fn iter(&self) -> impl Iterator<Item = (HazKey, Rect)> + '_ {
104        (0..self.keys.len()).map(|i| {
105            let bbox = Rect {
106                x_min: self.x_min[i],
107                y_min: self.y_min[i],
108                x_max: self.x_max[i],
109                y_max: self.y_max[i],
110            };
111            (self.keys[i], bbox)
112        })
113    }
114
115    /// Largest `x_max` of each chunk of 64 consecutive boxes.
116    pub fn chunk_x_max(&self) -> &[f32] {
117        &self.chunk_x_max
118    }
119
120    /// Calls `f` with the start and overlap mask of every chunk with a box overlapping `r`,
121    /// until it returns `true`. Returns whether it did.
122    #[inline(always)]
123    fn any_chunk(&self, r: Rect, mut f: impl FnMut(usize, u64) -> bool) -> bool {
124        // Boxes from `hi` on start right of `r`
125        let hi = if self.x_min.len() <= 64 {
126            // A single chunk: scanning it all is cheaper than searching
127            self.x_min.len()
128        } else {
129            self.x_min.partition_point(|&x| x <= r.x_max)
130        };
131        for (c, &x_max) in self.chunk_x_max.iter().enumerate() {
132            let start = c * 64;
133            if start >= hi {
134                break;
135            }
136            if x_max < r.x_min {
137                // Every box in the chunk ends left of `r`
138                continue;
139            }
140            let mask = self.chunk_mask(start, hi, r);
141            if mask != 0 && f(start, mask) {
142                return true;
143            }
144        }
145        false
146    }
147
148    /// Bit `j` is set if the hazard at `start + j` overlaps `r`, for up to 64 hazards before `hi`.
149    #[inline(always)]
150    fn chunk_mask(&self, start: usize, hi: usize, r: Rect) -> u64 {
151        let end = (start + 64).min(hi);
152        rect_mask(
153            &self.x_min[start..end],
154            &self.y_min[start..end],
155            &self.x_max[start..end],
156            &self.y_max[start..end],
157            r,
158        )
159    }
160}

162/// Bit `j` is set if box `j` overlaps `r` (borders included), for up to 64 boxes.
163#[inline(always)]
164#[allow(clippy::needless_bitwise_bool)] // branch-free
165pub fn rect_mask(x_min: &[f32], y_min: &[f32], x_max: &[f32], y_max: &[f32], r: Rect) -> u64 {
166    debug_assert!(x_min.len() <= 64);
167    let mut mask = 0;
168    for (j, (((&x0, &y0), &x1), &y1)) in x_min.iter().zip(y_min).zip(x_max).zip(y_max).enumerate() {
169        let overlaps = (x0 <= r.x_max) & (x1 >= r.x_min) & (y0 <= r.y_max) & (y1 >= r.y_min);
170        mask |= u64::from(overlaps) << j;
171    }
172    mask
173}
