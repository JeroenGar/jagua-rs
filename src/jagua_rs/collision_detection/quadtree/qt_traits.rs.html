9/// Common trait for all geometric primitives that can be directly queried in the quadtree
10/// for collisions with the edges of the registered hazards. These include: [Rect], [Edge] and [Circle].
11pub trait QTQueryable: CollidesWith<Edge> + CollidesWith<Rect> {
12    /// Checks
13    fn collides_with_quadrants(&self, r: &Rect, qs: [&Rect; 4]) -> [bool; 4] {
14        debug_assert!(r.quadrants().iter().zip(qs.iter()).all(|(q, r)| *q == **r));
15        qs.map(|q| self.collides_with(q))
16    }
17
18    /// Returns the preferred order for searching the quadrants around `split`.
19    /// The result must contain every index in `0..4` exactly once.
20    fn quadrant_order(&self, _split: Point) -> [usize; 4] {
21        [0, 1, 2, 3]
22    }
23}

25impl QTQueryable for Circle {
26    fn quadrant_order(&self, Point(mid_x, mid_y): Point) -> [usize; 4] {
27        let right = self.center.x() >= mid_x;
28        let top = self.center.y() >= mid_y;
29        let row_mask = usize::from(!top) << 1;
30        let containing_quadrant = row_mask | usize::from(right ^ top);
31        let first_adjacent_mask = row_mask | 1;
32        let order = [
33            containing_quadrant,
34            containing_quadrant ^ first_adjacent_mask,
35            containing_quadrant ^ (first_adjacent_mask ^ 2),
36            containing_quadrant ^ 2,
37        ];
38        debug_assert_eq!(
39            order,
40            match (right, top) {
41                (true, true) => [0, 1, 3, 2],
42                (false, true) => [1, 0, 2, 3],
43                (false, false) => [2, 1, 3, 0],
44                (true, false) => [3, 0, 2, 1],
45            }
46        );
47        order
48    }
49}

52impl QTQueryable for Edge {
53    #[allow(clippy::similar_names)]
54    fn collides_with_quadrants(&self, r: &Rect, qs: [&Rect; 4]) -> [bool; 4] {
55        debug_assert!(r.quadrants().iter().zip(qs.iter()).all(|(q, r)| *q == **r));
56        let e_x_min = self.x_min();
57        let e_x_max = self.x_max();
58        let e_y_min = self.y_min();
59        let e_y_max = self.y_max();
60
61        let [mut c_q0, mut c_q1, mut c_q2, mut c_q3] = [0, 1, 2, 3].map(|idx| {
62            let q = qs[idx];
63
64            let x_no_overlap = e_x_min.max(q.x_min) > e_x_max.min(q.x_max);
65            let y_no_overlap = e_y_min.max(q.y_min) > e_y_max.min(q.y_max);
66
67            if x_no_overlap || y_no_overlap {
68                // Edge is completely outside the x- or y-range of the quadrant
69                Some(false)
70            } else if q.collides_with(&self.start) || q.collides_with(&self.end) {
71                // Edge has at least one end point in the quadrant
72                Some(true)
73            } else {
74                // Undetermined, we need to check for intersections with the sides of the quadrants
75                None
76            }
77        });
78
79        // If all quadrants are already determined, we can return early
80        if let (Some(c_q0), Some(c_q1), Some(c_q2), Some(c_q3)) = (c_q0, c_q1, c_q2, c_q3) {
81            return [c_q0, c_q1, c_q2, c_q3];
82        }
83
84        // Otherwise, we need to check for intersections with the sides of the quadrants
85        // We can exploit the fact that the quadrants have a fixed layout, and share edges.
86
87        let c = r.centroid();
88
89        let [top, left, bottom, right] = r.sides();
90
91        let h_bisect = Edge {
92            start: Point(r.x_min, c.1),
93            end: Point(r.x_max, c.1),
94        };
95        let v_bisect = Edge {
96            start: Point(c.0, r.y_min),
97            end: Point(c.0, r.y_max),
98        };
99
100        //  1    0
101        //  2    3
102
103        half_intersect(self, &left, [&mut c_q1], [&mut c_q2]);
104        half_intersect(self, &right, [&mut c_q3], [&mut c_q0]);
105        half_intersect(self, &top, [&mut c_q0], [&mut c_q1]);
106        half_intersect(self, &bottom, [&mut c_q2], [&mut c_q3]);
107        half_intersect(
108            self,
109            &h_bisect,
110            [&mut c_q1, &mut c_q2],
111            [&mut c_q0, &mut c_q3],
112        );
113        half_intersect(
114            self,
115            &v_bisect,
116            [&mut c_q2, &mut c_q3],
117            [&mut c_q0, &mut c_q1],
118        );
119
120        let classified = [c_q0, c_q1, c_q2, c_q3].map(|collision| collision.unwrap_or(false));
121        #[cfg(debug_assertions)]
122        debug_assert!(
123            assertions::edge_quadrant_classification_is_conservative(self, qs, classified),
124            "edge: {self:?}, node: {r:?}, quadrants: {qs:?}, classified: {classified:?}, f64 oracle: {:?}",
125            assertions::edge_collisions_with_rects_f64(self, qs)
126        );
127
128        classified
129    }
130}

132/// If e1 intersects with e2 in the first half of e2, it sets all bools in `fst_qs` to true,
133/// if e1 intersects with e2 the second half of e2, it sets all bools in `sec_qs` to true.
134fn half_intersect<const N: usize>(
135    e1: &Edge,
136    e2: &Edge,
137    fst_qs: [&mut Option<bool>; N],
138    sec_qs: [&mut Option<bool>; N],
139) {
140    if fst_qs.iter().chain(sec_qs.iter()).any(|t| t.is_none())
141        && let Some((_, e2_col_loc)) = edge_intersection_half(e1, e2)
142    {
143        match e2_col_loc {
144            CollisionHalf::FirstHalf => {
145                for c in fst_qs {
146                    *c = Some(true);
147                }
148            }
149            CollisionHalf::Halfway => {
150                for c in fst_qs {
151                    *c = Some(true);
152                }
153                for c in sec_qs {
154                    *c = Some(true);
155                }
156            }
157            CollisionHalf::SecondHalf => {
158                for c in sec_qs {
159                    *c = Some(true);
160                }
161            }
162        }
163    }
164}

168// Similar to `edge_intersection`, but in case of an intersection, it returns in which half for both edge the intersection occurs.
169fn edge_intersection_half(e1: &Edge, e2: &Edge) -> Option<(CollisionHalf, CollisionHalf)> {
170    let Point(x1, y1) = e1.start;
171    let Point(x2, y2) = e1.end;
172    let Point(x3, y3) = e2.start;
173    let Point(x4, y4) = e2.end;
174
175    //based on: https://en.wikipedia.org/wiki/Line%E2%80%93line_intersection#Given_two_points_on_each_line_segment
176    let t_nom = (x2 - x4) * (y4 - y3) - (y2 - y4) * (x4 - x3);
177    let t_denom = (x2 - x1) * (y4 - y3) - (y2 - y1) * (x4 - x3);
178    let u_nom = (x2 - x4) * (y2 - y1) - (y2 - y4) * (x2 - x1);
179    let u_denom = (x2 - x1) * (y4 - y3) - (y2 - y1) * (x4 - x3);
180    if t_denom == 0.0 || u_denom == 0.0 {
181        //parallel edges
182        return None;
183    }
184
185    let t = t_nom / t_denom; //refers to the position along e1
186    let u = u_nom / u_denom; //refers to the position along e2
187    if (0.0..=1.0).contains(&t) && (0.0..=1.0).contains(&u) {
188        let e1_loc = match t.partial_cmp(&0.5).unwrap() {
189            Ordering::Greater => CollisionHalf::FirstHalf,
190            Ordering::Less => CollisionHalf::SecondHalf,
191            Ordering::Equal => CollisionHalf::Halfway,
192        };
193        let e2_loc = match u.partial_cmp(&0.5).unwrap() {
194            Ordering::Greater => CollisionHalf::FirstHalf,
195            Ordering::Less => CollisionHalf::SecondHalf,
196            Ordering::Equal => CollisionHalf::Halfway,
197        };
198
199        return Some((e1_loc, e2_loc));
200    }
201    None
202}
