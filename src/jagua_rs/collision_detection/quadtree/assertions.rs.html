3pub(super) fn edge_quadrant_classification_is_conservative(
4    edge: &Edge,
5    quadrants: [&Rect; 4],
6    classified: [bool; 4],
7) -> bool {
8    edge_collisions_with_rects_f64(edge, quadrants)
9        .into_iter()
10        .zip(classified)
11        .all(|(collides, classified)| !collides || classified)
12}

14pub(super) fn edge_collisions_with_rects_f64(edge: &Edge, rects: [&Rect; 4]) -> [bool; 4] {
15    rects.map(|rect| edge_collides_with_rect_f64(edge, rect))

18/// Recomputes the segment-rectangle intersection in `f64` so assertions do not inherit
19/// conservative rounding from the production `f32` predicate.
20#[allow(clippy::similar_names)]
21fn edge_collides_with_rect_f64(edge: &Edge, rect: &Rect) -> bool {
22    let (start_x, start_y) = (f64::from(edge.start.0), f64::from(edge.start.1));
23    let (end_x, end_y) = (f64::from(edge.end.0), f64::from(edge.end.1));
24    let (x_min, y_min) = (f64::from(rect.x_min), f64::from(rect.y_min));
25    let (x_max, y_max) = (f64::from(rect.x_max), f64::from(rect.y_max));
26
27    let x_no_overlap = start_x.min(end_x).max(x_min) > start_x.max(end_x).min(x_max);
28    let y_no_overlap = start_y.min(end_y).max(y_min) > start_y.max(end_y).min(y_max);
29    if x_no_overlap || y_no_overlap {
30        return false;
31    }
32
33    let point_in_rect = |x, y| x >= x_min && x <= x_max && y >= y_min && y <= y_max;
34    if point_in_rect(start_x, start_y) || point_in_rect(end_x, end_y) {
35        return true;
36    }
37
38    let edge_dx = end_x - start_x;
39    let edge_dy = end_y - start_y;
40    let sides = [
41        (x_max - start_x) * edge_dy - (y_max - start_y) * edge_dx,
42        (x_min - start_x) * edge_dy - (y_max - start_y) * edge_dx,
43        (x_min - start_x) * edge_dy - (y_min - start_y) * edge_dx,
44        (x_max - start_x) * edge_dy - (y_min - start_y) * edge_dx,
45    ];
46
47    !sides.iter().all(|side| *side > 0.0) && !sides.iter().all(|side| *side < 0.0)
48}
