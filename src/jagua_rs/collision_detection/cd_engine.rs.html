18/// The Collision Detection Engine (CDE).
19/// [`Hazard`]s can be (de)registered and collision queries can be performed.
20#[derive(Clone, Debug)]
21pub struct CDEngine {
22    /// Root node of the quadtree
23    pub quadtree: QTNode,
24    /// All hazards registered in the CDE (active and inactive)
25    pub hazards_map: SlotMap<HazKey, Hazard>,
26    /// Configuration of the CDE
27    pub config: CDEConfig,
28    /// The key of the hazard that represents the exterior of the container.
29    hkey_exterior: HazKey,
30}

32impl CDEngine {
33    #[must_use]
34    pub fn new(bbox: Rect, static_hazards: Vec<Hazard>, config: CDEConfig) -> CDEngine {
35        let mut quadtree = QTNode::new(config.quadtree_depth, bbox, config.cd_threshold);
36        let mut hazards_map = SlotMap::with_key();
37
38        for haz in static_hazards {
39            let hkey = hazards_map.insert(haz);
40            let qt_haz = QTHazard::from_root(quadtree.bbox, &hazards_map[hkey], hkey);
41            quadtree.register_hazard(qt_haz, &hazards_map);
42        }
43
44        let hkey_exterior = hazards_map
45            .iter()
46            .find(|(_, h)| matches!(h.entity, HazardEntity::Exterior))
47            .map(|(hkey, _)| hkey)
48            .expect("No exterior hazard registered in the CDE");
49
50        CDEngine {
51            quadtree,
52            hazards_map,
53            config,
54            hkey_exterior,
55        }
56    }

58    /// Registers a new hazard in the CDE.
59    pub fn register_hazard(&mut self, hazard: Hazard) {
60        debug_assert!(
61            !self.hazards_map.values().any(|h| h.entity == hazard.entity),
62            "Hazard with an identical entity already registered"
63        );
64        let hkey = self.hazards_map.insert(hazard);
65        let qt_hazard = QTHazard::from_root(self.bbox(), &self.hazards_map[hkey], hkey);
66        self.quadtree.register_hazard(qt_hazard, &self.hazards_map);
67
68        debug_assert!(assertions::qt_contains_no_dangling_hazards(self));
69    }

71    /// Removes a hazard from the CDE.
72    pub fn deregister_hazard_by_entity(&mut self, hazard_entity: HazardEntity) -> Hazard {
73        let hkey = self
74            .hazards_map
75            .iter()
76            .find(|(_, h)| h.entity == hazard_entity)
77            .map(|(hkey, _)| hkey)
78            .expect("Cannot deregister hazard that is not registered");
79
80        self.quadtree.deregister_hazard(hkey);
81        let hazard = self.hazards_map.remove(hkey).unwrap();
82        debug_assert!(assertions::qt_contains_no_dangling_hazards(self));
83
84        hazard
85    }
86
87    pub fn deregister_hazard_by_key(&mut self, hkey: HazKey) -> Hazard {
88        let hazard = self
89            .hazards_map
90            .remove(hkey)
91            .expect("Cannot deregister hazard that is not registered");
92        self.quadtree.deregister_hazard(hkey);
93        debug_assert!(assertions::qt_contains_no_dangling_hazards(self));
94
95        hazard
96    }

100        let dynamic_hazards = self
101            .hazards_map
102            .values()
103            .filter(|h| h.dynamic)
104            .cloned()
105            .collect_vec();
106        CDESnapshot { dynamic_hazards }

109    /// Restores the CDE to a previous state, as described by the snapshot.
110    pub fn restore(&mut self, snapshot: &CDESnapshot) {
111        //Restore the quadtree, by doing a 'diff' between the current state and the snapshot
112        //Only dynamic hazards are considered
113
114        //Determine which dynamic hazards need to be removed and which need to be added
115        let mut hazards_to_remove = self
116            .hazards_map
117            .iter()
118            .filter(|(_, h)| h.dynamic)
119            .map(|(hkey, h)| (hkey, h.entity))
120            .collect_vec();
121        let mut hazards_to_add = vec![];
122
123        for hazard in &snapshot.dynamic_hazards {
124            let present = hazards_to_remove
125                .iter()
126                .position(|(_, h)| h == &hazard.entity);
127            if let Some(idx) = present {
128                //the hazard is already present in the CDE, remove it from the hazards to remove
129                hazards_to_remove.swap_remove(idx);
130            } else {
131                //the hazard is not present in the CDE, add it to the list of hazards to add
132                hazards_to_add.push(hazard.clone());
133            }
134        }
135
136        //Remove all hazards currently in the CDE but not in the snapshot
137        for (hkey, _) in hazards_to_remove {
138            self.deregister_hazard_by_key(hkey);
139        }
140
141        //Add all hazards in the snapshot but not currently in the CDE
142        for hazard in hazards_to_add {
143            self.register_hazard(hazard);
144        }
145
146        debug_assert_eq!(
147            self.hazards_map.values().filter(|h| h.dynamic).count(),
148            snapshot.dynamic_hazards.len()
149        );
150    }

157    /// Checks whether a simple polygon collides with any of the (relevant) hazards
158    /// # Arguments
159    /// * `shape` - The shape (already transformed) to be checked for collisions
160    /// * `filter` - Hazard filter to be applied
161    pub fn detect_poly_collision(&self, shape: &SPolygon, filter: &impl HazardFilter) -> bool {
162        if self.bbox().relation_to(shape.bbox) == GeoRelation::Surrounding {
163            //Instead of each time starting from the quadtree root, we can use the virtual root (lowest level node which fully surrounds the shape)
164            let v_qt_root = self.get_virtual_root(shape.bbox);
165
166            // Check for edge intersections with the shape
167            for edge in shape.edge_iter() {
168                if v_qt_root.collides(&edge, filter).is_some() {
169                    return true;
170                }
171            }
172
173            // Check for containment of the shape in any of the hazards
174            for qt_hazard in v_qt_root.hazards.iter() {
175                match &qt_hazard.presence {
176                    QTHazPresence::None => {}
177                    QTHazPresence::Entire => unreachable!(
178                        "Entire hazards in the virtual root should have been caught by the edge intersection tests"
179                    ),
180                    QTHazPresence::Partial(_) => {
181                        if !filter.is_irrelevant(qt_hazard.hkey) {
182                            let haz_shape = &self.hazards_map[qt_hazard.hkey].shape;
183                            if self.detect_containment_collision(shape, haz_shape, qt_hazard.entity)
184                            {
185                                // The hazard is contained in the shape (or vice versa)
186                                return true;
187                            }
188                        }
189                    }
190                }
191            }
192
193            false
194        } else {
195            //The CDE does not capture the entire shape, so we can immediately return true
196            true
197        }
198    }

200    /// Checks whether a surrogate collides with any of the (relevant) hazards.
201    /// # Arguments
202    /// * `base_surrogate` - The (untransformed) surrogate to be checked for collisions
203    /// * `transform` - The transformation to be applied to the surrogate (on the fly)
204    /// * `filter` - Hazard filter to be applied
205    pub fn detect_surrogate_collision(
206        &self,
207        base_surrogate: &SPSurrogate,
208        transform: &Transformation,
209        filter: &impl HazardFilter,
210    ) -> bool {
211        for pole in base_surrogate.ff_poles() {
212            let t_pole = pole.transform_clone(transform);
213            if self.quadtree.collides(&t_pole, filter).is_some() {
214                return true;
215            }
216        }
217        for pier in base_surrogate.ff_piers() {
218            let t_pier = pier.transform_clone(transform);
219            if self.quadtree.collides(&t_pier, filter).is_some() {
220                return true;
221            }
222        }
223        false
224    }

226    /// Check for collision by containment between a shape and a hazard.
227    /// This only guarantees to detect collisions caused by full containment of one shape in another.
228    /// # Arguments
229    /// * `shape` - The shape to be checked for containment
230    /// * `haz_shape` - The shape of the respective hazard
231    /// * `haz_entity` - The entity inducing the hazard
232    #[must_use]
233    pub fn detect_containment_collision(
234        &self,
235        shape: &SPolygon,
236        haz_shape: &SPolygon,
237        haz_entity: HazardEntity,
238    ) -> bool {
239        //Due to possible fp issues, we check if the bboxes are "almost" related --
240        //meaning that, when edges are very close together, they are considered equal.
241        //Some relations which would normally be seen as `Intersecting` are now being considered `Enclosed`/`Surrounding` (which triggers the containment check).
242        let haz_to_shape_bbox_relation = haz_shape.bbox.almost_relation_to(shape.bbox);
243
244        //If the bounding boxes are contained, we have to check the actual shapes for containment.
245        //This can be done by testing whether a single point of the smaller shape is contained in the larger shape.
246        let contained = match haz_to_shape_bbox_relation {
247            GeoRelation::Surrounding => haz_shape.collides_with(&shape.poi.center),
248            GeoRelation::Enclosed => shape.collides_with(&haz_shape.poi.center),
249            GeoRelation::Disjoint | GeoRelation::Intersecting => false,
250        };
251
252        //Depending on the scope of the hazard this results a collision or not
253        match (haz_entity.scope(), contained) {
254            (GeoPosition::Interior, true) | (GeoPosition::Exterior, false) => true,
255            (GeoPosition::Interior, false) | (GeoPosition::Exterior, true) => false,
256        }
257    }

267    /// Collects polygon collisions until `stop_after_collision` requests an early return.
268    ///
269    /// The callback receives each colliding hazard's entity after it is added to the collector for
270    /// the first time. Returning `true` stops traversal and leaves the collector with only the
271    /// hazards found up to that point. Collision order is unspecified.
272    #[must_use]
273    pub fn collect_poly_collisions_until<C, F>(
274        &self,
275        shape: &SPolygon,
276        collector: &mut C,
277        mut stop_after_collision: F,
278    ) -> bool
279    where
280        C: HazardCollector,
281        F: FnMut(HazardEntity) -> bool,
282    {
283        if self.bbox().relation_to(shape.bbox) != GeoRelation::Surrounding
284            && !collector.contains_key(self.hkey_exterior)
285        {
286            collector.insert(self.hkey_exterior, HazardEntity::Exterior);
287            if stop_after_collision(HazardEntity::Exterior) {
288                return true;
289            }
290        }
291
292        //Instead of each time starting from the quadtree root, we can use the virtual root (lowest level node which fully surrounds the shape)
293        let v_quadtree = self.get_virtual_root(shape.bbox);
294
295        for edge in shape.edge_iter() {
296            if v_quadtree.collect_collisions_until(&edge, collector, &mut stop_after_collision) {
297                return true;
298            }
299        }
300
301        //Check if there are any other collisions due to containment
302        for qt_haz in v_quadtree.hazards.iter() {
303            match &qt_haz.presence {
304                // No need to check these, guaranteed to be detected by edge intersection
305                QTHazPresence::None | QTHazPresence::Entire => {}
306                QTHazPresence::Partial(_) => {
307                    if !collector.contains_key(qt_haz.hkey) {
308                        let h_shape = &self.hazards_map[qt_haz.hkey].shape;
309                        if self.detect_containment_collision(shape, h_shape, qt_haz.entity) {
310                            collector.insert(qt_haz.hkey, qt_haz.entity);
311                            if stop_after_collision(qt_haz.entity) {
312                                return true;
313                            }
314                        }
315                    }
316                }
317            }
318        }
319
320        false
321    }

335    /// Collects collisions found by the surrogate screening pass until
336    /// `stop_after_collision` requests an early return.
337    ///
338    /// The callback contract matches [`Self::collect_poly_collisions_until`]. The configured
339    /// fail-fast pole prefix is checked largest-first. This is not a complete polygon collision
340    /// query; follow it with [`Self::collect_poly_collisions_until`] if the callback does not stop
341    /// the screening pass.
342    #[must_use]
343    pub fn collect_surrogate_collisions_until<C, F>(
344        &self,
345        shape: &SPolygon,
346        collector: &mut C,
347        mut stop_after_collision: F,
348    ) -> bool
349    where
350        C: HazardCollector,
351        F: FnMut(HazardEntity) -> bool,
352    {
353        let Some(surrogate) = &shape.surrogate else {
354            return false;
355        };
356        for pole in surrogate.ff_poles() {
357            if self
358                .quadtree
359                .collect_collisions_until(pole, collector, &mut stop_after_collision)
360            {
361                return true;
362            }
363        }
364        false
365    }

367    /// Returns the lowest `QTNode` that completely surrounds the given bounding box.
368    /// Used to initiate collision checks from lower in the quadtree.
369    #[must_use]
370    pub fn get_virtual_root(&self, bbox: Rect) -> &QTNode {
371        let mut v_root = &self.quadtree;
372        while let Some(children) = v_root.children.as_ref() {
373            // Keep going down the tree until we cannot find a child that fully surrounds the shape
374            let surrounding_child = children
375                .iter()
376                .find(|child| child.bbox.relation_to(bbox) == GeoRelation::Surrounding);
377            match surrounding_child {
378                Some(child) => v_root = child,
379                None => break,
380            }
381        }
382        v_root
383    }
