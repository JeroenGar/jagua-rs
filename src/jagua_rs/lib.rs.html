10//!
11//! Collision detection for 2D irregular cutting and packing algorithms.
12//!
13//! [`geometry`] provides polygons and transformations. [`io`] imports item and container
14//! geometry and exports placements. A [`Layout`](entities::Layout) owns placements and
15//! keeps its [`CDEngine`](collision_detection::CDEngine) synchronized as items are added or removed.
16//! Use the CDE to query candidate placements before adding them to a layout.
17//! Optional `spp` and `bpp` features provide strip- and bin-packing models in [`probs`].
18//!
19//! # Import, query, place and restore
20//!
21//! ```
22//! use jagua_rs::collision_detection::{CDEConfig, hazards::filter::NoFilter};
23//! use jagua_rs::entities::Layout;
24//! use jagua_rs::geometry::{DTransformation, fail_fast::SPSurrogateConfig, geo_traits::Transformable};
25//! use jagua_rs::io::{ext_repr::{ExtContainer, ExtItem, ExtOrientation, ExtRotation, ExtShape}, import::Importer};
26//!
27//! # fn main() -> anyhow::Result<()> {
28//! let importer = Importer::new(CDEConfig {
29//!     quadtree_depth: 5,
30//!     cd_threshold: 96,
31//!     item_surrogate_config: SPSurrogateConfig::none(),
32//! }, None, None);
33//! let container = importer.import_container(&ExtContainer {
34//!     id: 0,
35//!     shape: ExtShape::Rectangle { x_min: 0.0, y_min: 0.0, width: 20.0, height: 10.0 },
36//!     zones: vec![],
37//! })?;
38//! let item = importer.import_item(&ExtItem {
39//!     id: 42,
40//!     shape: ExtShape::Rectangle { x_min: 0.0, y_min: 0.0, width: 2.0, height: 2.0 },
41//!     orientation: ExtOrientation {
42//!         rotation: ExtRotation::Discrete { angles: vec![0.0] },
43//!         reflection_axes: vec![],
44//!     },
45//!     min_quality: None,
46//! }, 0)?;
47//! let mut layout = Layout::new(container);
48//! let empty = layout.save();
49//!
50//! // Imported item geometry is centered at the origin; angles here are in radians.
51//! let pose = DTransformation::new(0.0, (5.0, 5.0));
52//! let candidate = item.shape_cd().transform_clone(&pose.compose());
53//! if !layout.cde().detect_poly_collision(&candidate, &NoFilter) {
54//!     // Placement itself does not validate collisions or orientation permissions.
55//!     layout.place_item(&item, pose);
56//! }
57//! assert_eq!(layout.placed_items().len(), 1);
58//! assert!(layout.is_collision_free());
59//! layout.restore(&empty)?;
60//! assert!(layout.placed_items().is_empty());
61//! # Ok(())
62//! # }
63//! ```
