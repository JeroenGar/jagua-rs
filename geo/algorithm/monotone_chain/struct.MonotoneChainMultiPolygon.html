pub struct MonotoneChainMultiPolygon<'a, T: GeoNum> { /* private fields */ }
Expand description

A MultiPolygon backed by MonotoneChain.

§Example

use geo::{MultiPolygon, Polygon, line_string, MonotoneChainMultiPolygon};

let polygon = Polygon::new(
    line_string![
        (x: 0., y: 0.),
        (x: 10., y: 0.),
        (x: 10., y: 10.),
        (x: 0., y: 10.),
        (x: 0., y: 0.),
    ],
    vec![]
);

let mp = MultiPolygon::new(vec![polygon.clone()]);
let m_mp:MonotoneChainMultiPolygon<f64> = (&mp).into();
